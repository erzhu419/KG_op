\section{Fisher-Whitened \texttt{SURE} Posterior and Eligible-Cascade Acquisition}\label{sec:model}\label{sec:vkg}

Within \textbf{GP-FTA}, a semiparametric target posterior combines shift-invariant within-policy variance estimation with closed-form Stein--Mallows (\texttt{SURE}) functional shrinkage and feeds our \emph{Eligible-Cascade Knowledge Gradient} (\textbf{EC-KG}) search rule and guarded three-slot verification cascade.

\subsection{Within-Policy Heteroscedastic Dispersion and Fisher-Whitened \texttt{SURE} Release}

When transferring to a shifted target task $t$ where $g_t(x) = \tilde{\phi}(x)^\top \mu_{0,g} + \Delta_g(x)$, fitting variance parameters to raw prior residuals $(Y_{t,g,\ell}(x) - \tilde{\phi}(x)^\top\mu_{0,g})^2$ leaks deterministic mean shift $\Delta_g(x)^2$ directly into aleatoric variance estimates, whereas plug-in leave-one-out GP selectors can retain misspecified source priors when sample sizes are small. \textbf{GP-FTA-ECKG} resolves both failure modes in closed form.

\paragraph{1. Shift-invariant within-policy HVD estimation.}
Given evaluated target policies $\mathcal{D}_n = \{x_1,\dots,x_{|\mathcal{D}_n|}\}$ with replicated subset $\mathcal{I}_{\mathrm{rep}} = \{i : r_i \ge 2\}$ and total within-policy degrees of freedom $\nu_n = \sum_{i \in \mathcal{I}_{\mathrm{rep}}} (r_i - 1)$, we estimate $\hat{\gamma}_{t,c} \in \R_+^{14}$ from centered within-policy dispersion:
\begin{equation}\label{eq:hvd_map}
  \hat{\gamma}_{t,c} = \arg\min_{\gamma \ge 0} \;
  \sum_{i \in \mathcal{I}_{\mathrm{rep}}} (r_i - 1)\left(\frac{S_{t,c,i}^2}{r_i - 1} - u_{\mathrm{HVD}}(x_i)^\top \gamma\right)^2
  + \lambda_{n,c}^{\mathrm{HVD}} \|\gamma - \gamma_{0,c}\|_2^2,
\end{equation}
where $\lambda_{n,c}^{\mathrm{HVD}} = (1 - \hat{\lambda}_{n,g}^{\mathrm{SURE}})\min\{3,\, 0.5 + 2.5 N_{\mathrm{src}}^{\mathrm{rep}}/36\} / (1 + 0.10 \nu_n)$ decays smoothly with both target degrees of freedom $\nu_n$ and \texttt{SURE} release weight $\hat{\lambda}_{n,g}^{\mathrm{SURE}}$ (switching to weak target-only regularization $0.5$ once released). Setting empirical median target dispersion $\hat{m}_{v,c} = \operatorname{med}_{i \in \mathcal{I}_{\mathrm{rep}}}(S_{t,c,i}^2/(r_i - 1))$, predictions use $\hat{v}_{n,c}(x) = \max\{u_{\mathrm{HVD}}(x)^\top \hat{\gamma}_{t,c},\, 10^{-10}\}$ and finite-sample upper confidence envelope $\hat{v}_{n,g}^{\mathrm{UCB}}(x) = \hat{v}_{n,g}(x)\big(1 + 0.35 / \sqrt{\max\{1, \nu_n\}}\big) + \min\{\underline{v}_{0,g},\, 0.20\,\hat{m}_{v,g}\}$.

\paragraph{2. Design-whitened Fisher-eigenspace \texttt{SURE} shrinkage and release.}
Let $y_{n,c} \in \R^{|\mathcal{D}_n|}$ collect observed policy sample means with precision matrix $W_{n,c} = \operatorname{diag}(r_i / \hat{v}_{n,c}(x_i))$ and normalized weights $\tilde{w}_{n,c} = W_{n,c}\mathbf{1} / (\mathbf{1}^\top W_{n,c}\mathbf{1})$. To separate global level shifts from functional shape transfer, we $W_{n,c}$-center active basis features $X_{n,c} = \Phi_{\mathrm{act},n} - \mathbf{1}\bar{\phi}_{n,c}^\top$ ($\bar{\phi}_{n,c} = \Phi_{\mathrm{act},n}^\top \tilde{w}_{n,c} \in \R^{K_{\mathrm{act}}}$, so $X_{n,c}^\top W_{n,c}\mathbf{1} = 0$) and form source-centered observations $\tilde{y}_{n,c} = (y_{n,c} - \bar{y}_{n,c}\mathbf{1}) - X_{n,c}\mu_{0,c,\mathrm{act}}$ ($\bar{y}_{n,c} = \tilde{w}_{n,c}^\top y_{n,c}$). Spectral decomposition of the target Fisher information matrix
\begin{equation}\label{eq:fisher_whitening}
  G_{n,c} = X_{n,c}^\top W_{n,c} X_{n,c} = Q_{n,c}\Lambda_{n,c}Q_{n,c}^\top
\end{equation}
identifies the target-identifiable eigenspace $\mathcal{I}_{n,c} = \{k \in \{1,\dots,K_{\mathrm{act}}\} : \Lambda_{n,c,k} > 10^{-6}\max\{1, \Lambda_{n,c,\max}\}\}$. Because $X_{n,c}^\top W_{n,c}\mathbf{1} = 0$, $W_{n,c}$-centering out $\bar{y}_{n,c}$ leaves the noise covariance along every identifiable eigenvector $q_{n,c,k}$ ($k \in \mathcal{I}_{n,c}$) equal to $G_{n,c}$ with zero approximation, yielding \emph{unit-variance, pairwise-orthogonal} whitened discrepancy projections:
\begin{equation}\label{eq:sure_closed_form}
  \begin{aligned}
    \tilde{R}_{n,c,k} &= \Lambda_{n,c,k}^{-1/2} q_{n,c,k}^\top X_{n,c}^\top W_{n,c} \tilde{y}_{n,c} = \delta_{n,c,k} + \xi_{n,c,k} \quad (\Var(\xi_{n,c,k}) = 1), \\
    \hat{\lambda}_{n,c}^{\mathrm{SURE}} &= \operatorname{clip}_{[0,1]}\!\left(1 - \frac{|\mathcal{I}_{n,c}|}{\sum_{k \in \mathcal{I}_{n,c}} \tilde{R}_{n,c,k}^2}\right),
  \end{aligned}
\end{equation}
with analogous 1-DOF Stein shrinkage $\hat{\lambda}_{n,c}^{\mathrm{int}} = \operatorname{clip}_{[0,1]}(1 - (\mathbf{1}^\top W_{n,c}\mathbf{1})^{-1}/(\bar{y}_{n,c} - \mu_{0,c,0} - \bar{\phi}_{n,c}^\top\mu_{0,c,\mathrm{act}})^2)$ on the global intercept. The resulting posterior GP mean and composite covariance blend source structure smoothly into target-only linear-plus-Mat\'ern estimation as $\hat{\lambda}_{n,c}^{\mathrm{SURE}} \to 1$:
\begin{align}
  m_{n,c}^{\mathrm{SURE}}(x) &= (1 - \hat{\lambda}_{n,c}^{\mathrm{int}})\mu_{0,c,0} + \hat{\lambda}_{n,c}^{\mathrm{int}}\big(\bar{y}_{n,c} - (1 - \hat{\lambda}_{n,c}^{\mathrm{SURE}})\bar{\phi}_{n,c}^\top\mu_{0,c,\mathrm{act}}\big) \notag\\
  &\quad + (1 - \hat{\lambda}_{n,c}^{\mathrm{SURE}})\tilde{\varphi}_{S_{\mathrm{act}}}(\psi(x))^\top\mu_{0,c,\mathrm{act}}, \label{eq:sure_mean}\\
  K_{n,c}^{\mathrm{SURE}}(x, x') &= (1 - \hat{\lambda}_{n,c}^{\mathrm{SURE}})\big(\tilde{\phi}(x)^\top \widehat{\Sigma}_{0,c}^{\mathrm{stab}}\tilde{\phi}(x') + w_{\perp,c}(x)^\top w_{\perp,c}(x')\big) \notag\\
  &\quad + \hat{\lambda}_{n,c}^{\mathrm{SURE}}\tau_c^2 \phi_{\mathrm{tgt}}(x)^\top\phi_{\mathrm{tgt}}(x') + \sigma_{\perp,c}^2 k_{5/2}(\tilde{\psi}_{\mathrm{ker}}(x), \tilde{\psi}_{\mathrm{ker}}(x'); \ell_c), \label{eq:sure_kernel}
\end{align}
where $(\tau_c^2, \sigma_{\perp,c}^2, \ell_c)$ are selected by marginal log-likelihood and $\tilde{\psi}_{\mathrm{ker}}$ switches to target-standardized $\psi$ once $\hat{\lambda}_{n,c}^{\mathrm{SURE}} \ge 0.50$. When $\max\{\hat{\lambda}_{n,f}^{\mathrm{SURE}}, \hat{\lambda}_{n,g}^{\mathrm{SURE}}\} \ge 0.65$, \textbf{GP-FTA-ECKG} marks the source prior released and disables source DCT template proposals in subsequent candidate pools, paired with balanced paid initialization ($N_0=8$: status-quo anchor, two top source templates, and five multi-channel DCT exploratory designs). For direct ablation against our predecessor selector, we also evaluate \texttt{GP-FTA-LOO-ECKG}, which retains the fixed-parameter leave-one-out predictive log-likelihood (\texttt{LOO-PLL}) branch comparison \citep[Eq.~5.12 of][]{rasmussen2006gaussian} on unshrunk $\Sigma_{0,c}^{\mathrm{raw}}$.

\subsection{Eligible-Cascade Knowledge Gradient (\texttt{EC-KG}) and Guarded Clean-Up}

\paragraph{Terminal cascade nomination and exact-binomial verification.}
At termination $N$, the nominator forms an ordered shortlist $\mathcal{S} = (x_{(0)}, \dots, x_{(L-1)}) \in \mathfrak{S}(\mathcal{D}_N)$ ($1 \le L \le 3$) strictly from evaluated policies $\mathcal{D}_N$: Slot~0 (\texttt{posterior\_primary}) maximizes exact-binomial gate-weighted expected saving $(\hat{J}_{\mathrm{rem},1} - \mu_{N,f}(x))_+ p_{N,0}^{\mathrm{gate}}(x)$ against the Bellman continuation value $\hat{J}_{\mathrm{rem},1}$ of Slots~1--2; Slot~1 (\texttt{empirical\_feasible\_incumbent}) selects the lowest-cost evaluated policy satisfying empirical safety margin $\bar{Y}_{t,g}(x) + z_{1-\alpha}\hat{s}_{\mathrm{pool}}(x) \le 0$ with preference for zero observed violations; and Slot~2 (\texttt{conservative\_posterior\_interior}) maximizes Slot-2 binomial gate power. After five-decimal deduplication, three-slot lists allocate fresh simulation blocks $(K_0, K_1, K_2) = (\SlotZeroK, \SlotOneK, \SlotTwoK)$ and error levels $(\delta_0, \delta_1, \delta_2) = (\SlotZeroDelta, \SlotOneDelta, \SlotTwoDelta)$ summing to $\delta = 0.05$ (critical violation count $k^\star = 1$ in every slot, so exact pass probability at violation rate $p$ is $C_{K_i,1}(p) = (1-p)^{K_i} + K_i p(1-p)^{K_i-1}$). When Slot~1 holds the incumbent-guard role, Slots~0 and~1 are both verified: if both pass, the policy with smaller verification sample mean objective is deployed; if exactly one passes, that policy is deployed; if both fail, Slot~2 is tested; and if no slot passes, the system abstains ($\hat{x}_t = \varnothing$, retaining $x_t^{\mathrm{sq}}$).

\paragraph{Two-stage Eligible-Cascade Knowledge Gradient (\texttt{EC-KG}).}
Single-slot lookahead (\texttt{GP-FTA-SingleSlotKG}) scores queries over a candidate subset $\mathcal{C}_n^{\mathrm{eval}}$ via $V_n(x') = (\hat{J}_n^{\mathrm{ref}} - \mu_{n,f}(x'))_+ p_{n,0}^{\mathrm{gate}}(x')$ using tower-aligned Gauss--Hermite quadrature (detailed in Electronic Companion Section~\ref*{ec:kg_correction}). Because terminal deployment actually executes the multi-slot guarded cascade over post-query evaluated policies, \textbf{GP-FTA-ECKG} directly values how a batch of $m_{\mathrm{batch}}$ replications at candidate $q \in \mathcal{C}_n$ improves the post-query nominated cascade over $\mathcal{D}_n \cup \{q\}$ in two fast vectorized stages:
\begin{enumerate}
  \item \emph{Stage~A (Eligible-pool pre-screen).} Vectorized rank-one updates across all $|\mathcal{C}_n| = 160$ candidate queries retain a contender set $\mathcal{K}_n$ ($|\mathcal{K}_n| = 16$) comprising the current shortlist policies, the single-slot gate-power and chance-feasibility winners, the top revisit candidates $q \in \mathcal{D}_n$, and the top unevaluated admission candidates $q \notin \mathcal{D}_n$.
  \item \emph{Stage~B (Correlated cascade nomination and dual switch-gain integration).} For each $q \in \mathcal{K}_n$, $S_{\mathrm{out}} = 16$ antithetic scrambled 4D Sobol draws generate hypothetical two-replication sufficient statistics $\xi^{(s)}$, updating posterior moments across $\mathcal{D}_n \cup \{q\}$ and running the exact nominator to obtain both the $\mathcal{D}_n$-restricted shortlist $\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n; q, \xi^{(s)})$ and the full post-query shortlist $\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n \cup \{q\}; q, \xi^{(s)})$ (identical whenever $q \in \mathcal{D}_n$). Conditional expected cascade saving $U_{n+1}(\mathcal{S}; q, \xi^{(s)})$—integrating joint constraint latents via $S_{\mathrm{in}} = 128$ common-random-number 3D Sobol draws and exact truncated-Gaussian guard expectations—is evaluated over nested menus $\mathfrak{M}_0^{(s)} = \{\varnothing, \mathcal{S}_n^\star, \hat{\mathcal{S}}_{n+1}(\mathcal{D}_n)\} \subseteq \mathfrak{M}_1^{(s)}(q) = \mathfrak{M}_0^{(s)} \cup \{\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n \cup \{q\})\}$, yielding
  \begin{equation}\label{eq:eckg_score}
    \widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q) = \underbrace{\frac{1}{S_{\mathrm{out}}}\sum_{s=1}^{S_{\mathrm{out}}} \Big(\max_{\mathcal{S} \in \mathfrak{M}_0^{(s)}} U_{n+1}(\mathcal{S}) - U_{n+1}(\mathcal{S}_n^\star)\Big)}_{\widehat{\mathrm{VOI}}_n^{\mathrm{refine}}(q) \,\ge\, 0}
    \;+\; \underbrace{\frac{1}{S_{\mathrm{out}}}\sum_{s=1}^{S_{\mathrm{out}}} \Big(\max_{\mathcal{S} \in \mathfrak{M}_1^{(s)}(q)} U_{n+1}(\mathcal{S}) - \max_{\mathcal{S} \in \mathfrak{M}_0^{(s)}} U_{n+1}(\mathcal{S})\Big)}_{\widehat{\mathrm{VOI}}_n^{\mathrm{admit}}(q) \,\ge\, 0}.
  \end{equation}
\end{enumerate}
Evaluating~\eqref{eq:eckg_score} requires $\sim 0.4$--$0.7\text{s}$ per online step (a $>200\times$ speedup over unscreened marginal-bracket cascade scoring), making native-eligibility multi-slot lookahead practical inside closed-loop search.
