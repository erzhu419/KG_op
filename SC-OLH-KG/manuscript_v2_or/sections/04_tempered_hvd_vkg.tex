\section{Online Shrinkage Posterior and Shortlist-Menu Cascade Lookahead}\label{sec:model}\label{sec:vkg}

On an unseen target task $t$, \textbf{GP-FTA} combines shift-invariant within-policy variance estimation with a Stein--Mallows (\texttt{SURE}) functional shrinkage rule inside a composite Gaussian process posterior, feeding a two-stage \emph{Eligible-Cascade Knowledge Gradient} (\textbf{EC-KG}) shortlist-menu lookahead proxy and an independent guarded three-slot clean-up cascade.

\subsection{Within-Policy Heteroscedastic Dispersion and Plug-In Fisher-Subspace \texttt{SURE}}

When transferring to a shifted target task $t$ where $g_t(x) = \tilde{\phi}(x)^\top \mu_{0,g} + \Delta_g(x)$, fitting variance parameters to raw prior residuals $(Y_{t,g,\ell}(x) - \tilde{\phi}(x)^\top\mu_{0,g})^2$ leaks deterministic mean shift $\Delta_g(x)^2$ into aleatoric variance estimates, whereas discrete leave-one-out GP branch selectors can retain misspecified source priors at small target sample sizes. \textbf{GP-FTA-ECKG} pairs constrained nonnegative least-squares (NNLS) within-policy variance estimation with a closed-form plug-in Fisher-subspace shrinkage rule.

\paragraph{1. Shift-invariant within-policy HVD estimation and variance buffer.}
Given evaluated target policies $\mathcal{D}_n = \{x_1,\dots,x_{|\mathcal{D}_n|}\}$ with replicated subset $\mathcal{I}_{\mathrm{rep}} = \{i : r_i \ge 2\}$ and total within-policy degrees of freedom $\nu_n = \sum_{i \in \mathcal{I}_{\mathrm{rep}}} (r_i - 1)$, we estimate $\hat{\gamma}_{t,c} \in \R_+^{14}$ strictly from centered within-policy dispersion:
\begin{equation}\label{eq:hvd_map}
  \hat{\gamma}_{t,c} = \arg\min_{\gamma \ge 0} \;
  \sum_{i \in \mathcal{I}_{\mathrm{rep}}} (r_i - 1)\left(\frac{S_{t,c,i}^2}{r_i - 1} - u_{\mathrm{HVD}}(x_i)^\top \gamma\right)^2
  + \lambda_{n,c}^{\mathrm{HVD}} \|\gamma - \gamma_{0,c}\|_2^2,
\end{equation}
where $\lambda_{n,c}^{\mathrm{HVD}} = (1 - \hat{\lambda}_{n,g}^{\mathrm{SURE}})\min\{3,\, 0.5 + 2.5 N_{\mathrm{src}}^{\mathrm{rep}}/36\} / (1 + 0.10 \nu_n)$ decays with target degrees of freedom $\nu_n$ and \texttt{SURE} shrinkage weight $\hat{\lambda}_{n,g}^{\mathrm{SURE}}$. Point variance predictions set $\hat{v}_{n,c}(x)=\max\{u_{\mathrm{HVD}}(x)^\top\hat{\gamma}_{t,c},10^{-10}\}$. Writing $m_g$ for the candidate-pool median constraint variance prediction, gate-power calculations apply a finite-degree-of-freedom buffer $\hat{v}_{n,g}^{\mathrm{buf}}(x) = \hat{v}_{n,g}(x)(1+0.35/\sqrt{\max\{1,\nu_n\}})+b_g$ ($b_g$ equal to the source residual floor for $\nu_n < 4$, the minimum of that floor and $0.20 m_g$ thereafter, or $0.05 m_g$ under target-only HVD; Section~\ref{sec:results} audits buffer sensitivity across $\{0.15, 0.35, 0.55\}$, while terminal safety guarantees rely strictly on fresh verification draws).

\paragraph{2. Plug-in Fisher-subspace \texttt{SURE} shrinkage and composite GP conditioning.}
Let $y_{n,c} \in \R^{|\mathcal{D}_n|}$ collect observed policy sample means ($|\mathcal{D}_n| \ge 4$) with plug-in precision matrix $\hat{W}_{n,c} = \operatorname{diag}(r_i / \hat{v}_{n,c}(x_i))$ and normalized weights $\tilde{w}_{n,c} = \hat{W}_{n,c}\mathbf{1} / (\mathbf{1}^\top \hat{W}_{n,c}\mathbf{1})$. To separate global level shifts from functional shape transfer, we $\hat{W}_{n,c}$-center active basis features $X_{n,c} = \Phi_{\mathrm{act},n} - \mathbf{1}\bar{\phi}_{n,c}^\top$ ($\bar{\phi}_{n,c} = \Phi_{\mathrm{act},n}^\top \tilde{w}_{n,c} \in \R^{K_{\mathrm{act}}}$, so $X_{n,c}^\top \hat{W}_{n,c}\mathbf{1} = 0$) and form source-centered observations $\tilde{y}_{n,c} = (y_{n,c} - \bar{y}_{n,c}\mathbf{1}) - X_{n,c}\mu_{0,c,\mathrm{act}}$ ($\bar{y}_{n,c} = \tilde{w}_{n,c}^\top y_{n,c}$). Spectral decomposition of the plug-in target Fisher matrix
\begin{equation}\label{eq:fisher_whitening}
  G_{n,c} = X_{n,c}^\top \hat{W}_{n,c} X_{n,c} = Q_{n,c}\Lambda_{n,c}Q_{n,c}^\top
\end{equation}
identifies the empirically identifiable eigenspace $\mathcal{I}_{n,c} = \{k \in \{1,\dots,K_{\mathrm{act}}\} : \Lambda_{n,c,k} > \max\{10^{-5}\operatorname{tr}(G_{n,c,+}),\, 10^{-8}\}\}$. Centering out $\bar{y}_{n,c}$ cancels any constant intercept shift identically; under a fixed design with exact precision $\hat{W}_{n,c} = V_{n,c}^{-1}$, projected errors have unit covariance $I_{|\mathcal{I}_{n,c}|}$. Motivated by Theorem~\ref{thm:sure_uniform}, we compute positive-part Stein--Mallows shrinkage weights
\begin{equation}\label{eq:sure_closed_form}
  \begin{aligned}
    \tilde{R}_{n,c,k} &= \Lambda_{n,c,k}^{-1/2} q_{n,c,k}^\top X_{n,c}^\top \hat{W}_{n,c} \tilde{y}_{n,c} \qquad (k \in \mathcal{I}_{n,c}), \\
    \hat{\lambda}_{n,c}^{\mathrm{SURE}} &= \operatorname{clip}_{[0,1]}\!\left(1 - \frac{|\mathcal{I}_{n,c}|}{\sum_{k \in \mathcal{I}_{n,c}} \tilde{R}_{n,c,k}^2}\right),
  \end{aligned}
\end{equation}
with analogous 1-DOF shrinkage $\hat{\lambda}_{n,c}^{\mathrm{int}} = \operatorname{clip}_{[0,1]}(1 - (\mathbf{1}^\top \hat{W}_{n,c}\mathbf{1})^{-1}/(\bar{y}_{n,c} - \mu_{0,c,0} - \bar{\phi}_{n,c}^\top\mu_{0,c,\mathrm{act}})^2)$ on the global intercept ($|\mathcal{D}_n| \ge 2$). These weights parameterize the composite GP prior prior to Gaussian conditioning:
\begin{align}
  m_{n,c}^{\mathrm{SURE}}(x) &= (1 - \hat{\lambda}_{n,c}^{\mathrm{int}})\mu_{0,c,0} + \hat{\lambda}_{n,c}^{\mathrm{int}}\big(\bar{y}_{n,c} - (1 - \hat{\lambda}_{n,c}^{\mathrm{SURE}})\bar{\phi}_{n,c}^\top\mu_{0,c,\mathrm{act}}\big) \notag\\
  &\quad + (1 - \hat{\lambda}_{n,c}^{\mathrm{SURE}})\tilde{\varphi}_{S_{\mathrm{act}}}(\psi(x))^\top\mu_{0,c,\mathrm{act}}, \label{eq:sure_mean}\\
  K_{n,c}^{\mathrm{SURE}}(x, x') &= (1 - \hat{\lambda}_{n,c}^{\mathrm{SURE}})\big(\tilde{\phi}(x)^\top \widehat{\Sigma}_{0,c}^{\mathrm{stab}}\tilde{\phi}(x') + w_{\perp,c}(x)^\top w_{\perp,c}(x')\big) \notag\\
  &\quad + \hat{\lambda}_{n,c}^{\mathrm{SURE}}\tau_c^2 \phi_{\mathrm{tgt}}(x)^\top\phi_{\mathrm{tgt}}(x') + \sigma_{\perp,c}^2 k_{5/2}(\tilde{\psi}_{\mathrm{ker}}(x), \tilde{\psi}_{\mathrm{ker}}(x'); \ell_c), \label{eq:sure_kernel}
\end{align}
where $(\tau_c^2, \sigma_{\perp,c}^2, \ell_c)$ are selected over a discrete marginal log-likelihood grid and $\tilde{\psi}_{\mathrm{ker}}$ switches to target-standardized $\psi$ once $\hat{\lambda}_{n,c}^{\mathrm{SURE}} \ge 0.50$. When $\max\{\hat{\lambda}_{n,f}^{\mathrm{SURE}}, \hat{\lambda}_{n,g}^{\mathrm{SURE}}\} \ge 0.65$, online candidate generation disables source DCT template proposals and switches HVD regularization to target-only while continuing to weight~\eqref{eq:sure_mean}--\eqref{eq:sure_kernel} smoothly by $1 - \hat{\lambda}_{n,c}^{\mathrm{SURE}}$. Full \textbf{GP-FTA-ECKG} pairs this posterior with balanced paid initialization ($N_0=8$: status-quo anchor, source DCT templates, and exploratory DCT designs); we also evaluate cold-initialized variants and a discrete leave-one-out predictive log-likelihood (\texttt{LOO-PLL}) selector \citep[Eq.~5.12 of][]{rasmussen2006gaussian}.

\subsection{Shortlist-Menu Lookahead (\texttt{EC-KG}) and Guarded Terminal Clean-Up}

\paragraph{Terminal cascade nomination, parameter-free alternative, and sequential curtailment.}
At termination $N$, the nominator forms an ordered shortlist $\hat{\mathcal{S}}_N(\mathcal{D}_N) = (x_{(0)}, \dots, x_{(L-1)})$ ($1 \le L \le 3$) from evaluated policies $\mathcal{D}_N$. In our structured backward-induction nominator, Slot~2 selects a conservative backup maximizing predicted gate power minus a small ($0.02$) normalized cost tie-breaker; Slot~1 selects an empirical incumbent minimizing blended sample and posterior cost among zero-violation candidates passing a pooled-variance margin and $0.55$ gate-power screen (falling back to minimum margin with zero-violation priority); and Slot~0 maximizes gate-weighted savings against continuation value among candidates with predicted gate power $\ge 0.55$ (falling back to best gate-weighted saving). Because these thresholds are engineering design choices, Section~\ref{sec:results} evaluates both an unthresholded terminal triple-utility maximizer (\texttt{TermOnly-UnthreshMaxU}) that maximizes $U_N(\mathcal{S})$~\eqref{eq:working_cascade_utility} over all distinct ordered triples of top evaluated candidates across guarded and unguarded modes on frozen terminal states $\mathcal{F}_N$ without $0.55$ gate thresholds or fixed Slot-1 rules, and an online restricted utility-based search/nomination variant (\texttt{RestrictedMaxU}).

Three-slot lists allocate fresh verification budgets $(K_0, K_1, K_2)$ and asymmetric error levels $(\delta_0, \delta_1, \delta_2) = (0.50\delta, 0.30\delta, 0.20\delta)$ summing to $\delta = 0.05$—set at baseline to $(\SlotZeroK, \SlotOneK, \SlotTwoK)$ ($k^\star = 1$ per slot, pass power $C_{K_i,1}(p) = (1-p)^{K_i} + K_i p(1-p)^{K_i-1}$), and evaluated across $k^\star \in \{0, 1, 2, 3\}$ in Section~\ref{sec:results}. When Slot~1 holds the incumbent-guard role, Slots~0 and~1 are verified: if both pass, the policy with lower verification sample mean cost is deployed; if one passes, it is deployed; if both fail, Slot~2 is tested; and if none passes, the system abstains ($\hat{x}_t = \varnothing$, retaining $x_t^{\mathrm{sq}}$). Moreover, by Theorem~\ref{thm:vkg_safety}(2), any slot can apply \emph{sequential binomial curtailment} \citep{wald1945sequential,batur2010finding}—halting immediately upon $k_i^\star + 1$ violations or $K_i - k_i^\star$ non-violations (completing full blocks only when both guarded slots pass)—to achieve identical certification outcomes in fewer replications.

\paragraph{Two-stage shortlist-menu lookahead proxy (\texttt{EC-KG}).}
Single-slot gate-power lookahead (\texttt{SingleSlotKG}) scores queries via $V_n(x') = (\hat{J}_n^{\mathrm{ref}} - \mu_{n,f}(x'))_+ p_{n,0}^{\mathrm{gate}}(x')$ using tower-aligned quadrature (Section~\ref*{ec:kg_correction}). To account for backup rescue, incumbent guarding, and sampled-policy eligibility $\mathcal{D}_n \cup \{q\}$, \textbf{EC-KG} evaluates a tractable \emph{nested shortlist-menu proxy} in two vectorized stages under the working GP surrogate (holding GP hyperparameters fixed across one-step fantasies and treating objective/constraint heads as conditionally independent):
\begin{enumerate}
  \item \emph{Stage~A (Eligible-pool pre-screen).} Vectorized rank-one updates across all $|\mathcal{C}_n| = 160$ candidates retain $|\mathcal{K}_n| \leq 16$ contenders combining current shortlist policies, single-slot gate-power and chance-feasibility winners, top revisits $q \in \mathcal{D}_n$, and top unevaluated admissions $q \notin \mathcal{D}_n$.
  \item \emph{Stage~B (Correlated cascade nomination and nested-menu integration).} Define the working-model expected clean-up saving of ordered shortlist $\mathcal{S}$ under conditional posterior $\mathcal{F}_m$ ($m \in \{n, n+1\}$) as
  \begin{equation}\label{eq:working_cascade_utility}
    U_m(\mathcal{S}) \coloneqq \E_m^{\mathrm{working}}\!\big[J_t(x_t^{\mathrm{sq}}) - \calL_{\mathrm{SQ}}(\text{clean-up outcome of }\mathcal{S})\big],
    \qquad U_m(\varnothing) \equiv 0,
  \end{equation}
  integrated across correlated slot constraint latents via $128$ common-random-number 3D Sobol draws and exact Gaussian Slot-0/Slot-1 guard expectations. For each $q \in \mathcal{K}_n$, $S_{\mathrm{out}}=8$ antithetic pairs of scrambled 4D Sobol draws ($16$ fantasies $\xi^{(s)}$) update posterior moments across $\mathcal{D}_n \cup \{q\}$ and form $\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n; q, \xi^{(s)})$ and $\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n \cup \{q\}; q, \xi^{(s)})$. Evaluating $U_{n+1}$ over nested menus $\mathfrak{M}_0^{(s)} = \{\varnothing, \mathcal{S}_n^\star, \hat{\mathcal{S}}_{n+1}(\mathcal{D}_n)\} \subseteq \mathfrak{M}_1^{(s)}(q) = \mathfrak{M}_0^{(s)} \cup \{\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n \cup \{q\})\}$ ($\mathcal{S}_n^\star = \hat{\mathcal{S}}_n(\mathcal{D}_n)$ if $U_n(\hat{\mathcal{S}}_n(\mathcal{D}_n)) > 0$, else $\varnothing$) yields
  \begin{equation}\label{eq:eckg_score}
    \widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q) = \underbrace{\frac{1}{2 S_{\mathrm{out}}}\sum_{s=1}^{2 S_{\mathrm{out}}} \Big(\max_{\mathcal{S} \in \mathfrak{M}_0^{(s)}} U_{n+1}(\mathcal{S}) - U_{n+1}(\mathcal{S}_n^\star)\Big)}_{\widehat{\mathrm{VOI}}_n^{\mathrm{refine}}(q) \,\ge\, 0}
    \;+\; \underbrace{\frac{1}{2 S_{\mathrm{out}}}\sum_{s=1}^{2 S_{\mathrm{out}}} \Big(\max_{\mathcal{S} \in \mathfrak{M}_1^{(s)}(q)} U_{n+1}(\mathcal{S}) - \max_{\mathcal{S} \in \mathfrak{M}_0^{(s)}} U_{n+1}(\mathcal{S})\Big)}_{\widehat{\mathrm{VOI}}_n^{\mathrm{admit}}(q) \,\ge\, 0}.
  \end{equation}
\end{enumerate}
Crucially,~\eqref{eq:eckg_score} is a \emph{menu-maximization proxy} rather than exact single-nomination value because terminal clean-up deploys $\hat{\mathcal{S}}_N(\mathcal{D}_N)$ directly without maximizing over $\{\varnothing, \mathcal{S}_{N-1}^\star\}$. Holding $(\tau_c^2, \sigma_{\perp,c}^2, \ell_c, \hat{\lambda}_{n,c}^{\mathrm{SURE}})$ fixed across one-step fantasies follows standard conjugate KG approximations \citep{frazier2009correlated,scott2011correlated}, omitting second-order hyperparameter-learning value, while conditional head independence given $\psi(x)$ captures shared operational trends through $\psi(x)$ without modeling residual objective--constraint shock correlation. Sections~\ref{sec:theory} and~\ref{sec:results} formalize and audit both Stage~A screening loss and menu-versus-nomination discrepancies.
