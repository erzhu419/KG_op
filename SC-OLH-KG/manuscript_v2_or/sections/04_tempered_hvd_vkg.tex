\section{Online Shrinkage Posterior and Shortlist-Menu Cascade Lookahead}\label{sec:model}\label{sec:vkg}

On an unseen target task $t$, \textbf{GP-FTA} combines shift-invariant within-policy variance estimation with a Stein--Mallows (\texttt{SURE}) functional shrinkage rule inside a composite Gaussian process posterior, feeding a two-stage \emph{Eligible-Cascade Knowledge Gradient} (\textbf{EC-KG}) shortlist-menu lookahead proxy and an independent guarded three-slot clean-up cascade.

\subsection{Within-Policy Heteroscedastic Dispersion and Plug-In Fisher-Subspace \texttt{SURE}}

When transferring to a shifted target task $t$ where $g_t(x) = \tilde{\phi}(x)^\top \mu_{0,g} + \Delta_g(x)$, fitting variance parameters to raw prior residuals $(Y_{t,g,\ell}(x) - \tilde{\phi}(x)^\top\mu_{0,g})^2$ leaks deterministic mean shift $\Delta_g(x)^2$ into aleatoric variance estimates, whereas discrete leave-one-out GP branch selectors can retain misspecified source priors at small target sample sizes. \textbf{GP-FTA-ECKG} addresses both issues via two closed-form online updates.

\paragraph{1. Shift-invariant within-policy HVD estimation and heuristic variance buffer.}
Given evaluated target policies $\mathcal{D}_n = \{x_1,\dots,x_{|\mathcal{D}_n|}\}$ with replicated subset $\mathcal{I}_{\mathrm{rep}} = \{i : r_i \ge 2\}$ and total within-policy degrees of freedom $\nu_n = \sum_{i \in \mathcal{I}_{\mathrm{rep}}} (r_i - 1)$, we estimate $\hat{\gamma}_{t,c} \in \R_+^{14}$ from centered within-policy dispersion:
\begin{equation}\label{eq:hvd_map}
  \hat{\gamma}_{t,c} = \arg\min_{\gamma \ge 0} \;
  \sum_{i \in \mathcal{I}_{\mathrm{rep}}} (r_i - 1)\left(\frac{S_{t,c,i}^2}{r_i - 1} - u_{\mathrm{HVD}}(x_i)^\top \gamma\right)^2
  + \lambda_{n,c}^{\mathrm{HVD}} \|\gamma - \gamma_{0,c}\|_2^2,
\end{equation}
where $\lambda_{n,c}^{\mathrm{HVD}} = (1 - \hat{\lambda}_{n,g}^{\mathrm{SURE}})\min\{3,\, 0.5 + 2.5 N_{\mathrm{src}}^{\mathrm{rep}}/36\} / (1 + 0.10 \nu_n)$ decays with target degrees of freedom $\nu_n$ and \texttt{SURE} shrinkage weight $\hat{\lambda}_{n,g}^{\mathrm{SURE}}$ (switching to weak target-only regularization $0.5$ once the source-release threshold is crossed). Point variance predictions use $\hat{v}_{n,c}(x)=\max\{u_{\mathrm{HVD}}(x)^\top\hat{\gamma}_{t,c},10^{-10}\}$. Write $m_g$ for the median constraint variance prediction in the current candidate batch. The main method scores gate power using $\hat{v}_{n,g}(x)(1+0.35/\sqrt{\max\{1,\nu_n\}})+b_g$. With source HVD active, $b_g$ equals the source residual floor until four target degrees of freedom, then the smaller of that floor and $0.20m_g$. With target-only HVD, $b_g=0.05m_g$. Section~\ref{sec:results} reports empirical coverage. Terminal safety follows from the independent verification draws.

\paragraph{2. Plug-in Fisher-subspace \texttt{SURE} shrinkage and composite GP conditioning.}
Let $y_{n,c} \in \R^{|\mathcal{D}_n|}$ collect observed policy sample means ($|\mathcal{D}_n| \ge 4$) with plug-in precision matrix $\hat{W}_{n,c} = \operatorname{diag}(r_i / \hat{v}_{n,c}(x_i))$ and normalized weights $\tilde{w}_{n,c} = \hat{W}_{n,c}\mathbf{1} / (\mathbf{1}^\top \hat{W}_{n,c}\mathbf{1})$. To separate global level shifts from functional shape transfer, we $\hat{W}_{n,c}$-center active basis features $X_{n,c} = \Phi_{\mathrm{act},n} - \mathbf{1}\bar{\phi}_{n,c}^\top$ ($\bar{\phi}_{n,c} = \Phi_{\mathrm{act},n}^\top \tilde{w}_{n,c} \in \R^{K_{\mathrm{act}}}$, so $X_{n,c}^\top \hat{W}_{n,c}\mathbf{1} = 0$) and form source-centered observations $\tilde{y}_{n,c} = (y_{n,c} - \bar{y}_{n,c}\mathbf{1}) - X_{n,c}\mu_{0,c,\mathrm{act}}$ ($\bar{y}_{n,c} = \tilde{w}_{n,c}^\top y_{n,c}$). Spectral decomposition of the plug-in target Fisher matrix
\begin{equation}\label{eq:fisher_whitening}
  G_{n,c} = X_{n,c}^\top \hat{W}_{n,c} X_{n,c} = Q_{n,c}\Lambda_{n,c}Q_{n,c}^\top
\end{equation}
identifies the empirically identifiable eigenspace $\mathcal{I}_{n,c} = \{k \in \{1,\dots,K_{\mathrm{act}}\} : \Lambda_{n,c,k} > \max\{10^{-5}\operatorname{tr}(G_{n,c,+}),\, 10^{-8}\}\}$. Because $X_{n,c}^\top \hat{W}_{n,c}\mathbf{1} = 0$, centering out $\bar{y}_{n,c}$ cancels any constant intercept shift identically; if the design is fixed and $\hat{W}_{n,c}$ equals the true inverse error covariance $V_{n,c}^{-1}$, the projected error covariance $A_{n,c} V_{n,c} A_{n,c}^\top$ ($A_{n,c} = \Lambda_{n,c,\mathcal{I}}^{-1/2} Q_{n,c,\mathcal{I}}^\top X_{n,c}^\top \hat{W}_{n,c}$) equals $I_{|\mathcal{I}_{n,c}|}$ (whereas under plug-in $\hat{v}_{n,c} \neq v_{t,c}$ and adaptive sampling, unit variance is an empirical plug-in approximation). Motivated by the idealized reference bound in Theorem~\ref{thm:sure_uniform}, we compute plug-in Stein--Mallows shrinkage weights
\begin{equation}\label{eq:sure_closed_form}
  \begin{aligned}
    \tilde{R}_{n,c,k} &= \Lambda_{n,c,k}^{-1/2} q_{n,c,k}^\top X_{n,c}^\top \hat{W}_{n,c} \tilde{y}_{n,c} \qquad (k \in \mathcal{I}_{n,c}), \\
    \hat{\lambda}_{n,c}^{\mathrm{SURE}} &= \operatorname{clip}_{[0,1]}\!\left(1 - \frac{|\mathcal{I}_{n,c}|}{\sum_{k \in \mathcal{I}_{n,c}} \tilde{R}_{n,c,k}^2}\right),
  \end{aligned}
\end{equation}
with analogous 1-DOF shrinkage $\hat{\lambda}_{n,c}^{\mathrm{int}} = \operatorname{clip}_{[0,1]}(1 - (\mathbf{1}^\top \hat{W}_{n,c}\mathbf{1})^{-1}/(\bar{y}_{n,c} - \mu_{0,c,0} - \bar{\phi}_{n,c}^\top\mu_{0,c,\mathrm{act}})^2)$ on the global intercept ($|\mathcal{D}_n| \ge 2$). These weights parameterize the prior mean and composite GP covariance prior to Gaussian conditioning:
\begin{align}
  m_{n,c}^{\mathrm{SURE}}(x) &= (1 - \hat{\lambda}_{n,c}^{\mathrm{int}})\mu_{0,c,0} + \hat{\lambda}_{n,c}^{\mathrm{int}}\big(\bar{y}_{n,c} - (1 - \hat{\lambda}_{n,c}^{\mathrm{SURE}})\bar{\phi}_{n,c}^\top\mu_{0,c,\mathrm{act}}\big) \notag\\
  &\quad + (1 - \hat{\lambda}_{n,c}^{\mathrm{SURE}})\tilde{\varphi}_{S_{\mathrm{act}}}(\psi(x))^\top\mu_{0,c,\mathrm{act}}, \label{eq:sure_mean}\\
  K_{n,c}^{\mathrm{SURE}}(x, x') &= (1 - \hat{\lambda}_{n,c}^{\mathrm{SURE}})\big(\tilde{\phi}(x)^\top \widehat{\Sigma}_{0,c}^{\mathrm{stab}}\tilde{\phi}(x') + w_{\perp,c}(x)^\top w_{\perp,c}(x')\big) \notag\\
  &\quad + \hat{\lambda}_{n,c}^{\mathrm{SURE}}\tau_c^2 \phi_{\mathrm{tgt}}(x)^\top\phi_{\mathrm{tgt}}(x') + \sigma_{\perp,c}^2 k_{5/2}(\tilde{\psi}_{\mathrm{ker}}(x), \tilde{\psi}_{\mathrm{ker}}(x'); \ell_c), \label{eq:sure_kernel}
\end{align}
where $(\tau_c^2, \sigma_{\perp,c}^2, \ell_c)$ are selected over a discrete marginal log-likelihood grid and $\tilde{\psi}_{\mathrm{ker}}$ switches to target-standardized $\psi$ once $\hat{\lambda}_{n,c}^{\mathrm{SURE}} \ge 0.50$. When $\max\{\hat{\lambda}_{n,f}^{\mathrm{SURE}}, \hat{\lambda}_{n,g}^{\mathrm{SURE}}\} \ge 0.65$, the model sets its \emph{template/HVD release flag} (\texttt{is\_source\_released}), disabling source DCT template proposals in subsequent candidate pools and switching HVD regularization to target-only, while~\eqref{eq:sure_mean}--\eqref{eq:sure_kernel} continue to weight source functional terms smoothly by $1 - \hat{\lambda}_{n,c}^{\mathrm{SURE}}$ (so crossing $0.65$ indicates threshold release rather than exact zero source contribution unless $\hat{\lambda}_{n,c}^{\mathrm{SURE}} = 1$). Full \textbf{GP-FTA-ECKG} pairs this posterior with balanced paid initialization ($N_0=8$: status-quo anchor, two top source templates, and five cold DCT designs); we also evaluate cold-initialized variants (\texttt{ColdInit-MetaPrior-SURE-ECKG}) to estimate the aggregate contribution of the warm configuration, as well as \texttt{GP-FTA-LOO-ECKG}, which replaces~\eqref{eq:finite_source_cov} and~\eqref{eq:sure_closed_form} with the discrete leave-one-out predictive log-likelihood (\texttt{LOO-PLL}) branch selector \citep[Eq.~5.12 of][]{rasmussen2006gaussian} on $\Sigma_{0,c}$.

\subsection{Shortlist-Menu Lookahead (\texttt{EC-KG}) and Guarded Terminal Clean-Up}

\paragraph{Terminal cascade nomination and exact-binomial verification.}
At termination $N$, the nominator forms a single ordered shortlist $\hat{\mathcal{S}}_N(\mathcal{D}_N) = (x_{(0)}, \dots, x_{(L-1)}) \in \mathfrak{S}(\mathcal{D}_N)$ ($1 \le L \le 3$) from evaluated policies $\mathcal{D}_N$. Backward induction starts from a conservative third-slot candidate ranked by predicted gate power with a small objective penalty. The empirical incumbent for Slot~1 minimizes a blended empirical and predicted cost among zero-violation candidates passing a pooled-variance margin and a $0.55$ gate-power threshold. If this set is empty the code minimizes the margin, giving zero-violation records priority. The primary favors gate-weighted savings relative to continuation when its estimated gate power is at least $0.55$. Otherwise the fallback rules use predicted cost and gate power. After five-decimal deduplication, three-slot lists allocate fresh simulation blocks $(K_0, K_1, K_2) = (\SlotZeroK, \SlotOneK, \SlotTwoK)$ and error levels $(\delta_0, \delta_1, \delta_2) = (\SlotZeroDelta, \SlotOneDelta, \SlotTwoDelta)$ summing to $\delta = 0.05$ (critical violation count $k^\star = 1$ in every slot, so pass probability at violation rate $p$ is $C_{K_i,1}(p) = (1-p)^{K_i} + K_i p(1-p)^{K_i-1}$). When Slot~1 holds the incumbent-guard role, Slots~0 and~1 are both verified: if both pass, the policy with smaller verification sample mean objective is deployed; if exactly one passes, that policy is deployed; if both fail, Slot~2 is tested; and if no slot passes, the system abstains ($\hat{x}_t = \varnothing$, retaining $x_t^{\mathrm{sq}}$).

\paragraph{Two-stage shortlist-menu lookahead proxy (\texttt{EC-KG}).}
Single-slot lookahead (\texttt{GP-FTA-SingleSlotKG}) scores queries over a candidate subset via $V_n(x') = (\hat{J}_n^{\mathrm{ref}} - \mu_{n,f}(x'))_+ p_{n,0}^{\mathrm{gate}}(x')$ using tower-aligned Gauss--Hermite quadrature (Section~\ref*{ec:kg_correction}; note that \texttt{SingleSlotKG} and \texttt{FeasibleKG} also differ slightly in their candidate pre-screening rule unless matched screening is enforced). To account for multi-slot backup rescue, objective-guard comparison, and last-query eligibility $\mathcal{D}_n \cup \{q\}$, \textbf{EC-KG} evaluates a tractable \emph{nested shortlist-menu lookahead proxy} in two vectorized stages under the working GP surrogate (assuming independent objective/constraint GP heads, Gaussian constraint-violation mapping with $\hat{v}_{n,g}^{\mathrm{buf}}$, and fixed GP hyperparameters within one-step fantasies):
\begin{enumerate}
  \item \emph{Stage~A (Eligible-pool pre-screen).} Vectorized rank-one updates across all $|\mathcal{C}_n| = 160$ candidate queries retain a contender set $\mathcal{K}_n$ ($|\mathcal{K}_n| \leq 16$ in online search) comprising the current shortlist policies, the single-slot gate-power and chance-feasibility winners, the top revisit candidates $q \in \mathcal{D}_n$, and the top unevaluated admission candidates $q \notin \mathcal{D}_n$.
  \item \emph{Stage~B (Correlated cascade nomination and nested-menu integration).} For each contender $q \in \mathcal{K}_n$, $S_{\mathrm{out}}=8$ antithetic pairs of scrambled four-dimensional Sobol points ($16$ fantasy evaluations $\xi^{(s)}$) update posterior moments across $\mathcal{D}_n \cup \{q\}$ and invoke the terminal nominator to obtain both the $\mathcal{D}_n$-restricted shortlist $\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n; q, \xi^{(s)})$ and the post-query shortlist $\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n \cup \{q\}; q, \xi^{(s)})$ (identical whenever $q \in \mathcal{D}_n$). Working-model conditional expected cascade saving $U_{n+1}(\mathcal{S}; q, \xi^{(s)})$—integrating joint constraint latents via $128$ common-random-number 3D Sobol draws and exact Gaussian guard expectations—is evaluated over nested shortlist menus $\mathfrak{M}_0^{(s)} = \{\varnothing, \mathcal{S}_n^\star, \hat{\mathcal{S}}_{n+1}(\mathcal{D}_n)\} \subseteq \mathfrak{M}_1^{(s)}(q) = \mathfrak{M}_0^{(s)} \cup \{\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n \cup \{q\})\}$, where $\mathcal{S}_n^\star = \hat{\mathcal{S}}_n(\mathcal{D}_n)$ if $U_n(\hat{\mathcal{S}}_n(\mathcal{D}_n)) > 0$ and $\varnothing$ otherwise:
  \begin{equation}\label{eq:eckg_score}
    \widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q) = \underbrace{\frac{1}{2 S_{\mathrm{out}}}\sum_{s=1}^{2 S_{\mathrm{out}}} \Big(\max_{\mathcal{S} \in \mathfrak{M}_0^{(s)}} U_{n+1}(\mathcal{S}) - U_{n+1}(\mathcal{S}_n^\star)\Big)}_{\widehat{\mathrm{VOI}}_n^{\mathrm{refine}}(q) \,\ge\, 0}
    \;+\; \underbrace{\frac{1}{2 S_{\mathrm{out}}}\sum_{s=1}^{2 S_{\mathrm{out}}} \Big(\max_{\mathcal{S} \in \mathfrak{M}_1^{(s)}(q)} U_{n+1}(\mathcal{S}) - \max_{\mathcal{S} \in \mathfrak{M}_0^{(s)}} U_{n+1}(\mathcal{S})\Big)}_{\widehat{\mathrm{VOI}}_n^{\mathrm{admit}}(q) \,\ge\, 0}.
  \end{equation}
\end{enumerate}
Crucially,~\eqref{eq:eckg_score} is a \emph{menu-maximization lookahead proxy} rather than the exact one-step value of actual terminal deployment, because actual clean-up at step~$N$ deploys $\hat{\mathcal{S}}_N(\mathcal{D}_N)$ directly without maximizing $U_N$ over retained historical menus $\{\varnothing, \mathcal{S}_{N-1}^\star\}$. Section~\ref{sec:theory} formalizes this distinction, and Section~\ref{sec:results} examines screening, finite-grid sensitivity, and agreement between the menu proxy and single-nomination score on frozen search states using independent $256 \times 1{,}024$ all-$160$ reference evaluations.
