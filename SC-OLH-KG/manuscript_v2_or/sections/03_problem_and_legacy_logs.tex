\section{Certification Contract, Operational Summaries, and Spectral Priors}\label{sec:prior}

\subsection{Certification Contract, Operating Risk, and Public Operational Summaries}\label{sec:domain_invariant_psi}

Consider $M_{\mathrm{src}}$ historical source tasks $m \in \{1,\dots,M_{\mathrm{src}}\}$ and an unseen target task $t$. Each task $k \in \{1,\dots,M_{\mathrm{src}}\} \cup \{t\}$ governs $C_k \ge 1$ interleaved operational control channels over planning horizon or subnetwork size $H_k$, yielding normalized policy space $\cX_k = [0,1]^{d_k}$ with $d_k = C_k H_k$ (or $d_k$ variable signal phases), where $d_m \neq d_t$ across horizons or subnetworks. Simulating policy $x \in \cX_k$ for $r \ge 1$ independent replications under private exogenous shock law $\mathbb{Q}_k$ produces paired observations
\begin{equation}\label{eq:obs_model}
  Y_{k,f,\ell}(x) = J_k(x) + \varepsilon_{k,f,\ell}(x),
  \qquad
  Y_{k,g,\ell}(x) = g_k(x) + \varepsilon_{k,g,\ell}(x)
  \qquad (\ell = 1,\dots,r),
\end{equation}
with zero-mean noise $\E[\varepsilon_{k,c,\ell}(x)] = 0$, state-dependent variances $v_{k,c}(x) = \Var_{\mathbb{Q}_k}(Y_{k,c,\ell}(x))$ for heads $c \in \{f, g\}$, and true violation probability $p_k(x) \coloneqq \Prob_{\mathbb{Q}_k}(Y_{k,g,\ell}(x) > 0)$, where $Y_{k,g,\ell}(x)$ is centered at the operational threshold $\tau_k$ so positive values denote violations. On target task $t$, we seek
\begin{equation}\label{eq:main_ccp}
  \min_{x \in \cX_t} \; J_t(x)
  \qquad \text{subject to} \qquad p_t(x) \le \alpha,
\end{equation}
at target risk level $\alpha \in (0, 1)$ ($\alpha = 0.05$ in our experiments).

Each target task specifies a pre-existing domain status-quo anchor $x_t^{\mathrm{sq}} \in \cX_t$, paid search budget $B_{\mathrm{search}} = (N_0 + N) m_{\mathrm{batch}}$ ($N_0 = 8$ initial designs, $N$ online steps, $m_{\mathrm{batch}} = 2$ replications per step), and independent terminal verification cap $B_{\mathrm{ver}}$. After search history $\mathcal{F}_N$ is frozen, terminal clean-up \citep{boesel2003using} outputs either a certified deployment ($C$, deploying $\hat{x}_t \in \mathcal{D}_N$, where $\mathcal{D}_N \subset \cX_t$ denotes the finite set of policies actually simulated during search) or a strict abstention ($C^c$, $\hat{x}_t = \varnothing$).

\paragraph{Operational rationale for sampled-policy eligibility ($\hat{x}_t \in \mathcal{D}_N$).}
Because terminal clean-up draws fresh simulation trajectories independent of $\mathcal{F}_N$, exact binomial gates control false certification at level $\delta$ for \emph{any} fixed candidate—including an unvisited policy predicted solely by the surrogate. Restricting terminal nomination to $\mathcal{D}_N$ reflects an operational and statistical governance convention rather than a requirement of Theorem~\ref{thm:vkg_safety}: first, unvisited policies ($r_i = 0$) carry zero direct observations, so centered within-policy dispersion $S_i^2/(r_i - 1)$ and empirical violation counts $V_i$ are unobserved and nomination must rely entirely on extrapolated surrogate noise predictions; second, change-control protocols in service, grid, and traffic engineering commonly require candidate schedules to pass preliminary simulator sanity screening during search before consuming dedicated certification budgets \citep{scott2011correlated,ungredda2024bayesian}. In Section~\ref{sec:results}, we test relaxing $\hat{x}_t \in \mathcal{D}_N$ directly by allowing unvisited surrogate optima into terminal shortlists.

The deployed operating policy is $X_t^{\mathrm{op}} = \hat{x}_t$ on $C$ and $X_t^{\mathrm{op}} = x_t^{\mathrm{sq}}$ on $C^c$, incurring status-quo decision loss
\begin{equation}\label{eq:sq_loss}
  \calL_{\mathrm{SQ}}(C, \hat{x}_t) \coloneqq J_t(X_t^{\mathrm{op}}) =
  \begin{cases}
    J_t(\hat{x}_t), & \text{if certified } (C), \\
    J_t(x_t^{\mathrm{sq}}), & \text{if abstained } (C^c).
  \end{cases}
\end{equation}
Conditioned on $\mathcal{F}_N$, the probability of operating an infeasible policy under verification law $\mathbb{Q}_t^{\mathrm{ver}}$ decomposes into certified false acceptance and retained status-quo non-compliance:
\begin{equation}\label{eq:unsafe_operation}
  \Prob\big(p_t^{\mathrm{ver}}(X_t^{\mathrm{op}}) > \alpha \,\big|\, \mathcal{F}_N\big)
  = \Prob\big(C \cap \{p_t^{\mathrm{ver}}(\hat{x}_t) > \alpha\} \,\big|\, \mathcal{F}_N\big)
  + \mathbb{I}\{p_t^{\mathrm{ver}}(x_t^{\mathrm{sq}}) > \alpha\}\,\Prob(C^c \mid \mathcal{F}_N).
\end{equation}
Theorem~\ref{thm:vkg_safety} bounds the first term by $\delta = 0.05$ under $\mathbb{Q}_t^{\mathrm{ver}}$. When the status-quo anchor is feasible ($p_t^{\mathrm{ver}}(x_t^{\mathrm{sq}}) \le \alpha$, as in our inventory and queueing benchmarks),~\eqref{eq:unsafe_operation} guarantees $\Prob(p_t^{\mathrm{ver}}(X_t^{\mathrm{op}}) > \alpha \mid \mathcal{F}_N) \le \delta$. Conversely, when $x_t^{\mathrm{sq}}$ itself violates the chance constraint ($p_t^{\mathrm{ver}}(x_t^{\mathrm{sq}}) > \alpha$, as in our urban signal retiming networks), abstention avoids certifying an unsafe challenger but retains status-quo non-compliance unless search certifies a feasible replacement. If post-deployment conditions shift away from $\mathbb{Q}_t^{\mathrm{ver}}$ (as between disjoint historical power-grid block pools), certified policies remain protected on $\mathbb{Q}_t^{\mathrm{ver}}$ but can exhibit out-of-sample exceedances, which we audit explicitly. Alongside $\calL_{\mathrm{SQ}}$, we report $\calL_{10\%} = J_t(\hat{x}_t)\mathbb{I}_C + 1.10 J_t(x_t^{\mathrm{sq}})\mathbb{I}_{C^c}$, compliance-penalized operating loss $\calL_{\mathrm{comp}}(X_t^{\mathrm{op}}; \lambda_{\mathrm{pen}}) = J_t(X_t^{\mathrm{op}}) + \lambda_{\mathrm{pen}}|J_t(x_t^{\mathrm{sq}})|\max\{p_t(X_t^{\mathrm{op}}) - \alpha, 0\}$ ($\lambda_{\mathrm{pen}} = 2$ baseline, with sensitivity across $\lambda_{\mathrm{pen}} \in \{1, 2, 5, 10\}$), and optimality gap relative to a high-budget feasible reference benchmark.

\paragraph{Public Nominal Rollouts ($\psi \in \R^{\PsiDim}$), Reconstructible Domain Specification, and Generic Alternative.}
Prior to paid simulation on task $k$, the learner observes only \emph{public planning inputs} $\cI_k^{\mathrm{pub}}$ (published demand or arrival forecasts, storage ratings, signal phase bounds)—never private shock laws $\mathbb{Q}_k$. Deterministic rollout of $x \in [0,1]^{d_k}$ under $\cI_k^{\mathrm{pub}}$ produces a nominal trajectory $\cT_k^{\mathrm{nom}}(x)$ and dimension-invariant exposure bundle
\begin{equation}\label{eq:nominal_exposure}
  x \in [0,1]^{d_k} \;\longmapsto\; \cT_k^{\mathrm{nom}}(x) \;\longmapsto\;
  \Xi_k(x) = \big(\psi(x),\, A_{\mathrm{local}}(x),\, N_{\mathrm{shared}}(x)\big) \in \R^{\PsiDim} \times \R_+^{4} \times \R_+^{3},
\end{equation}
where $\psi(x) \in \R^{\PsiDim}$ combines seven normalized trajectory statistics with $(A_{\mathrm{local},1}(x), A_{\mathrm{local},2}(x))$ and $N_{\mathrm{shared}}(x) \in \R_+^3$, while $A_{\mathrm{local}}(x)$ and $N_{\mathrm{shared}}(x)$ parameterize local and three-regime (tertile) simulation volatility. Importantly, $\Xi_k(x)$ is a \emph{predetermined domain-engineered interface} rather than a learned representation; offline prior learning operates on top of $\Xi_k(x)$ within each domain (and in Section~\ref{sec:results} we also evaluate a simulator-free black-box control-profile summary $\psi^{\mathrm{generic}}(x)$ that uses no domain fluid equations).

To illustrate~\eqref{eq:nominal_exposure} concretely, consider dual-sourcing inventory control over horizon $H_k$ ($d_k = 2 H_k$) with public demand forecast $\hat{d}_{1:H_k} > 0$, warehouse capacity $C_{\mathrm{cap}}$, initial stock $I_0 \in [0, C_{\mathrm{cap}}]$, pipeline $P_0 = 0$, nominal supplier yield $\bar{\eta} = 0.90$, expedited cap $C_{\mathrm{exp}} = 0.35 C_{\mathrm{cap}}$, normalized forecast $\tilde{d}_h = \hat{d}_h / \bar{d}$ ($\bar{d} = H_k^{-1}\sum_{h=1}^{H_k}\hat{d}_h$, $D_{\mathrm{tot}} = H_k \bar{d}$), and forecast tertile regimes $\mathcal{R}_1, \mathcal{R}_2, \mathcal{R}_3 \subset \{1,\dots,H_k\}$. Policy $x \in [0,1]^{2 H_k}$ sets regular order-up-to target $S_h^{\mathrm{reg}} = (0.10 + 0.85 x_{2h-1}) C_{\mathrm{cap}}$ and expedited trigger $S_h^{\mathrm{exp}} = 0.70 x_{2h} S_h^{\mathrm{reg}}$, generating deterministic recursion ($h = 1,\dots,H_k$):
\begin{equation}\label{eq:inv_nominal_recursion}
  \begin{aligned}
    a_h^{\mathrm{pre}} &= \min\{C_{\mathrm{cap}},\, I_{h-1} + \bar{\eta} P_{h-1}\}, \quad
    q_h^{\mathrm{exp}} = \min\{\max\{S_h^{\mathrm{exp}} - a_h^{\mathrm{pre}},\, 0\},\, C_{\mathrm{exp}}\}, \quad
    a_h = \min\{C_{\mathrm{cap}},\, a_h^{\mathrm{pre}} + q_h^{\mathrm{exp}}\}, \\
    s_h &= \max\{\hat{d}_h - a_h,\, 0\}, \quad
    I_h = \max\{a_h - \hat{d}_h,\, 0\}, \quad
    q_h^{\mathrm{reg}} = \max\{\min\{S_h^{\mathrm{reg}} - I_h,\, C_{\mathrm{cap}} - I_h\},\, 0\}, \quad
    P_h = q_h^{\mathrm{reg}},
  \end{aligned}
\end{equation}
with post-demand buffer $b_h = I_h / \max\{\hat{d}_h, 10^{-6}\}$ and lean-stock exposure $\ell_h = \exp(-1.5 b_h)$. Table~\ref{tab:psi_inventory} defines all entries of $(\psi(x), A_{\mathrm{local}}(x), N_{\mathrm{shared}}(x))$; normalizing by $H_k$, $D_{\mathrm{tot}}$, $C_{\mathrm{cap}}$, or $|\mathcal{R}_r|$ renders policies across $H_s \in \{36, 48\}$ and $H_t \in \{96, 120\}$ commensurate, while compressing fine period-by-period timing into summary ratios ($d_k \gg \PsiDim$). Electronic Companion Section~\ref*{ec:psi_specs} records the corresponding recursions for queueing, European grid storage, urban signal timing, and the generic black-box profile summary.

\paragraph{Within-Policy Sufficient Statistics.}
For every batch of $r \ge 1$ replications at policy $x$, we record sample means $\bar{Y}_{k,c}(x) = r^{-1}\sum_{\ell=1}^r Y_{k,c,\ell}(x)$, centered within-policy sums of squares $S_{k,c}^2(x) = \sum_{\ell=1}^r (Y_{k,c,\ell}(x) - \bar{Y}_{k,c}(x))^2$ ($S_{k,c}^2(x) \equiv 0$ when $r = 1$), and violation counts $V_k(x) = \sum_{\ell=1}^r \mathbb{I}\{Y_{k,g,\ell}(x) > 0\}$, merging repeat visits to identical policies via parallel variance formulas.

\subsection{Offline Spectral Prior Extraction and Structural Inductives}

Given an offline archive $\mathcal{D}_{\mathrm{src}} = \{(m_a, x_a, r_a, \bar{Y}_{f,a}, S_{f,a}^2, \bar{Y}_{g,a}, S_{g,a}^2, \Xi_{m_a}(x_a))\}_{a=1}^{N_{\mathrm{src}}}$ logged across $M_{\mathrm{src}}$ source tasks by adaptive legacy solvers, offline prior construction combines four structural mechanisms over a prescribed feature dictionary (with all tuning constants summarized in Electronic Companion Table~\ref*{tab:ec_hyperparams} and ablated individually in Section~\ref{sec:results}):
\begin{enumerate}
  \item \textbf{Empirical Inverse-Occupancy Balancing.} Standardizing $\tilde{\psi}_a = \operatorname{diag}(\hat{\sigma}_\psi)^{-1}(\psi_a - \hat{\mu}_\psi)$ and forming Gaussian kernel affinities $W_{ab} = \exp(-\|\tilde{\psi}_a - \tilde{\psi}_b\|_2^2 / (2 h_{\mathrm{occ}}^2))$ with empirical occupancy $\hat{\pi}_a = N_{\mathrm{src}}^{-1}\sum_{b=1}^{N_{\mathrm{src}}} W_{ab}$, we compute self-normalized clipped weights ($\bar{w} = 6$)
  \begin{equation}\label{eq:inv_occ_weights}
    w_a = \frac{\min\{\bar{w},\; \operatorname{med}(\hat{\pi}^{-1})^{-1}\hat{\pi}_a^{-1}\}}{\sum_{b=1}^{N_{\mathrm{src}}} \min\{\bar{w},\; \operatorname{med}(\hat{\pi}^{-1})^{-1}\hat{\pi}_b^{-1}\}},
    \qquad \hat{\nu}_w = \sum_{a=1}^{N_{\mathrm{src}}} w_a \delta_{\tilde{\psi}_a},
  \end{equation}
  mitigating dense solver-cluster dominance on observed archive support without claiming to recover unvisited regions of $\mathcal{M}_\psi$.

  \item \textbf{Prescribed Low-Order Dictionary and Graph-Laplacian Smoothing.} Standardized coordinates are clipped componentwise to $\bar{\psi}_j = \operatorname{clip}(\tilde{\psi}_j, -4, 4)$. Over $\bar{\psi} \in [-4,4]^{\PsiDim}$, a $\RawDictDim$-feature dictionary $\varphi(\bar{\psi})$ combines linear terms $\bar{\psi}_j$, normalized second-order Hermite polynomials $(\bar{\psi}_j^2 - 1)/\sqrt{2}$, bounded trigonometric modes $\sin(\frac{\pi}{2}\operatorname{clip}(\bar{\psi}_j/2.5, -1, 1))$, and action--regime cross-interactions $\bar{\psi}_j \bar{\psi}_k$. Using normalized graph Laplacian $L_{\mathrm{sym}} = I - \Lambda_{\mathrm{deg}}^{-1/2} W \Lambda_{\mathrm{deg}}^{-1/2}$, features above the $0.75$ quantile of Dirichlet roughness $\bar{\varphi}_j^\top L_{\mathrm{sym}} \bar{\varphi}_j / (\|\bar{\varphi}_j\|_2^2 + \epsilon_0)$ are filtered out ($39$ smooth dictionary features survive).

  \item \textbf{Gram Whitening, Sparse Mode Selection, and Two-Stage Covariance Shrinkage.} Centering $\mathring{\varphi} = \varphi_{\mathrm{lp}} - \E_{\hat{\nu}_w}[\varphi_{\mathrm{lp}}]$ and eigendecomposing $G_w = \sum_a w_a \mathring{\varphi}_a \mathring{\varphi}_a^\top = U\Sigma_{\mathrm{gram}} U^\top$, truncated whitening $Q_{\mathrm{orth}} = \Sigma_{\mathrm{gram},+}^{-1/2}U_+^\top$ gives unclipped archive projections $\varphi_a^{\mathrm{orth}} = Q_{\mathrm{orth}}\mathring{\varphi}_a$ unit Gram matrix under $\hat{\nu}_w$. Per-task ridge-GLS fits $(\hat{\theta}_{m,f}, \hat{\theta}_{m,g})$ weighted by $w_a r_a$ regress on unclipped $\varphi_a^{\mathrm{orth}}$, while target feature evaluation clips $\tilde{\varphi}(\psi) = \operatorname{clip}(Q_{\mathrm{orth}}\mathring{\varphi}(\psi), -5, 5)$ to bound extrapolation. Modes are ranked by cross-task signal-to-dispersion $\sum_{c \in \{f,g\}} \omega_c \big(\bar{\theta}_{k,c}^2 / (\Var_m(\hat{\theta}_{m,k,c}) + \epsilon_0) + \tfrac{1}{2}\E_m[\hat{\theta}_{m,k,c}^2]\big)$ ($\omega_f = 1, \omega_g = 1.5$), retaining active rank $K_{\mathrm{act}} = \ActiveRank$ ($\tilde{\phi}(x) = [1, \tilde{\varphi}_{S_{\mathrm{act}}}(\psi(x))^\top]^\top \in \R^{p}$, $p = 9$). Setting $\mu_{0,c} = \bar{\theta}_{S_{\mathrm{act}},c}$ and sample covariance $\Sigma_{0,c}^{\mathrm{emp}} = \widehat{\operatorname{Cov}}_m(\hat{\theta}_{m,S_{\mathrm{act}},c})$, offline fitting applies meta-mean inflation and diagonal stabilization ($s_0 = 0.15$), followed online by dimension-aware shrinkage $\omega_{\mathrm{src}} = p / (M_{\mathrm{src}} + p)$:
  \begin{equation}\label{eq:finite_source_cov}
    \Sigma_{0,c} = \Big(1 + \frac{1}{M_{\mathrm{src}}}\Big)\big((1 - s_0)\Sigma_{0,c}^{\mathrm{emp}} + s_0 D_{0,c}\big) + 10^{-4} I_p,
    \qquad
    \widehat{\Sigma}_{0,c}^{\mathrm{stab}} = (1 - \omega_{\mathrm{src}})\,\Sigma_{0,c} + \omega_{\mathrm{src}}\,\operatorname{diag}(\Sigma_{0,c}).
  \end{equation}
  Pruned modes $k \in S_{\mathrm{act}}^c$ form energy-weighted GP features $w_{\perp,c}(x) = \tfrac{1}{2}\,\tilde{\varphi}_{S_{\mathrm{act}}^c}(\psi(x)) \odot \sqrt{\hat{e}_{\perp,c}}$.

  \item \textbf{Multi-Replication HVD Hyperprior and Channel-Preserving DCT Templates.} Over the $14$-feature nonnegative basis $u_{\mathrm{HVD}}(x) = [1,\, A_{\mathrm{local}}(x)^{\odot 2},\, N_{\mathrm{shared}}(x)^{\odot 2},\, \{\tfrac{1}{2}(N_i(x)+N_j(x))^2\}_{1 \le i < j},\, N_{\mathrm{shared}}(x)]^\top \in \R_+^{14}$, degrees-of-freedom-weighted NNLS fits $\gamma_{0,c}$ strictly on multi-replication records $\{a : r_a \ge 2\}$. Elite feasible source policies are compressed per channel via orthonormal DCT-II into $K_{\mathrm{dct}} = 12$ smooth modes to provide horizon-invariant warm-start templates.
\end{enumerate}
