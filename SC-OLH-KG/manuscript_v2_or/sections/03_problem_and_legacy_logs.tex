\section{Problem Contract and Four-Pillar Functional Meta-Prior}\label{sec:prior}

\subsection{Roster-Invariant Decision Contract, Operating Risk, and Public Rollouts}

Consider $M_{\mathrm{src}}$ historical source tasks $m \in \{1,\dots,M_{\mathrm{src}}\}$ and an unseen target task $t$. Each task $k \in \{1,\dots,M_{\mathrm{src}}\} \cup \{t\}$ governs $C_k \in \{1, 2\}$ interleaved operational control channels over planning horizon or subnetwork size $H_k$, yielding normalized policy space $\cX_k = [0,1]^{d_k}$ with $d_k = C_k H_k$ (or $d_k$ variable signal phases), where $d_m \neq d_t$ across horizons or subnetworks. Simulating policy $x \in \cX_k$ for $r \ge 1$ independent replications under private exogenous shock law $\mathbb{Q}_k$ produces paired observations
\begin{equation}\label{eq:obs_model}
  Y_{k,f,\ell}(x) = J_k(x) + \varepsilon_{k,f,\ell}(x),
  \qquad
  Y_{k,g,\ell}(x) = g_k(x) + \varepsilon_{k,g,\ell}(x)
  \qquad (\ell = 1,\dots,r),
\end{equation}
with zero-mean noise $\E[\varepsilon_{k,c,\ell}(x)] = 0$, state-dependent variances $v_{k,c}(x) = \Var_{\mathbb{Q}_k}(Y_{k,c,\ell}(x))$ for heads $c \in \{f, g\}$, and true violation probability $p_k(x) \coloneqq \Prob_{\mathbb{Q}_k}(Y_{k,g,\ell}(x) > 0)$, where $Y_{k,g,\ell}(x)$ is centered at the operational threshold $\tau_k$ so positive values denote violations. On target task $t$, we seek
\begin{equation}\label{eq:main_ccp}
  \min_{x \in \cX_t} \; J_t(x)
  \qquad \text{subject to} \qquad p_t(x) \le \alpha,
\end{equation}
at target risk level $\alpha \in (0, 1)$ (fixed at $\alpha = 0.05$ in our experiments).

Each target task specifies a pre-existing domain status-quo anchor $x_t^{\mathrm{sq}} \in \cX_t$, paid search budget $B_{\mathrm{search}} = (N_0 + N) m_{\mathrm{batch}}$ ($N_0 = 8$ initial designs, $N$ online steps, $m_{\mathrm{batch}} = 2$ replications per step), and independent terminal verification cap $B_{\mathrm{ver}} = \MaxVerifCalls$. After search history $\mathcal{F}_N$ is frozen, terminal verification outputs either a certified deployment ($\mathrm{state} = \texttt{certified\_deployment}$, $\hat{x}_t \in \mathcal{D}_N$, where $\mathcal{D}_N \subset \cX_t$ denotes the finite set of policies actually simulated during search) or a strict abstention ($\mathrm{state} = \texttt{abstention}$, $\hat{x}_t = \varnothing$). The deployed operating policy is $X_t^{\mathrm{op}} = \hat{x}_t$ on the certification event $C$ and $X_t^{\mathrm{op}} = x_t^{\mathrm{sq}}$ on $C^c$, incurring roster-invariant status-quo decision loss
\begin{equation}\label{eq:sq_loss}
  \calL_{\mathrm{SQ}}(\mathrm{state}, \hat{x}_t) \coloneqq J_t(X_t^{\mathrm{op}}) =
  \begin{cases}
    J_t(\hat{x}_t), & \text{if } \mathrm{state} = \texttt{certified\_deployment}, \\
    J_t(x_t^{\mathrm{sq}}), & \text{if } \mathrm{state} = \texttt{abstention}.
  \end{cases}
\end{equation}
Conditioned on frozen search history $\mathcal{F}_N$, the probability of operating an infeasible policy under verification law $\mathbb{Q}_t^{\mathrm{ver}}$ decomposes into certified false acceptance and retained status-quo non-compliance:
\begin{equation}\label{eq:unsafe_operation}
  \Prob\big(p_t^{\mathrm{ver}}(X_t^{\mathrm{op}}) > \alpha \,\big|\, \mathcal{F}_N\big)
  = \Prob\big(C \cap \{p_t^{\mathrm{ver}}(\hat{x}_t) > \alpha\} \,\big|\, \mathcal{F}_N\big)
  + \mathbb{I}\{p_t^{\mathrm{ver}}(x_t^{\mathrm{sq}}) > \alpha\}\,\Prob(C^c \mid \mathcal{F}_N).
\end{equation}
Theorem~\ref{thm:vkg_safety} bounds the first term by the familywise verification level $\delta = 0.05$ under $\mathbb{Q}_t^{\mathrm{ver}}$. When the status-quo anchor is feasible ($p_t^{\mathrm{ver}}(x_t^{\mathrm{sq}}) \le \alpha$, as indicated by the independent audits of our inventory and queueing anchors),~\eqref{eq:unsafe_operation} guarantees $\Prob(p_t^{\mathrm{ver}}(X_t^{\mathrm{op}}) > \alpha \mid \mathcal{F}_N) \le \delta$. Conversely, when the legacy anchor $x_t^{\mathrm{sq}}$ itself violates the chance constraint ($p_t^{\mathrm{ver}}(x_t^{\mathrm{sq}}) > \alpha$, as in our target traffic networks), abstention avoids false certification of a new candidate but continues operating an unsafe status quo; a certified replacement can restore compliance. If post-deployment operating conditions shift away from $\mathbb{Q}_t^{\mathrm{ver}}$ (as in the comparison between disjoint, interleaved historical power-grid block pools), certified deployments remain protected on $\mathbb{Q}_t^{\mathrm{ver}}$ but can experience out-of-distribution exceedances, which we audit explicitly. Alongside $\calL_{\mathrm{SQ}}$, we report $\calL_{10\%}(\mathrm{state}, \hat{x}_t) = J_t(\hat{x}_t)\mathbb{I}_C + 1.10 J_t(x_t^{\mathrm{sq}})\mathbb{I}_{C^c}$ and compliance-penalized operating loss $\calL_{\mathrm{comp}}(X_t^{\mathrm{op}}) = J_t(X_t^{\mathrm{op}}) + 2|J_t(x_t^{\mathrm{sq}})|\max\{p_t(X_t^{\mathrm{op}}) - \alpha, 0\}$.

\paragraph{Public Nominal Rollout ($\psi \in \R^{\PsiDim}$) and Fiber Geometry.}
Prior to paid simulation on task $k$, the learner observes only \emph{public planning inputs} $\cI_k^{\mathrm{pub}}$ (published demand or arrival forecasts, storage ratings, signal phase bounds)—never private shock laws $\mathbb{Q}_k$ or realized forecast errors. Deterministic rollout of $x \in [0,1]^{d_k}$ under $\cI_k^{\mathrm{pub}}$ produces a nominal trajectory $\cT_k^{\mathrm{nom}}(x)$ and dimension-invariant exposure bundle
\begin{equation}\label{eq:nominal_exposure}
  x \in [0,1]^{d_k} \;\longmapsto\; \cT_k^{\mathrm{nom}}(x) \;\longmapsto\;
  \Xi_k(x) = \big(\psi(x),\, A_{\mathrm{local}}(x),\, N_{\mathrm{shared}}(x)\big) \in \R^{\PsiDim} \times \R_+^{4} \times \R_+^{3},
\end{equation}
where $\psi(x) \in \R^{\PsiDim}$ summarizes domain-engineered nominal utilization, buffer headroom, throughput, and regime deficits; $A_{\mathrm{local}}(x) \in \R_+^4$ records root-mean-square local volatility exposures; and $N_{\mathrm{shared}}(x) \in \R_+^3$ measures off-peak, shoulder, and peak regime occupancies. Unit tests across all four benchmark families verify bit-exact invariance ($\|\Delta\psi\|_\infty = 0$) when private shock parameters are perturbed. Because $d_k \gg \PsiDim$, $\psi$ is non-injective: two raw policies with identical $\psi$ receive the same latent GP mean and covariance predictions given a fixed history. Their $A_{\mathrm{local}}$ and $N_{\mathrm{shared}}$ exposures can differ, which can change their simulation variance estimates and predicted gate power. Section~\ref{sec:results} reports measured changes in $\psi$ and simulator outputs for raw-policy perturbations.

\paragraph{Within-Policy Sufficient Statistics.}
For every batch of $r \ge 1$ replications at policy $x$, we record sample means $\bar{Y}_{k,c}(x) = r^{-1}\sum_{\ell=1}^r Y_{k,c,\ell}(x)$, centered within-policy sums of squares $S_{k,c}^2(x) = \sum_{\ell=1}^r (Y_{k,c,\ell}(x) - \bar{Y}_{k,c}(x))^2$ ($S_{k,c}^2(x) \equiv 0$ when $r = 1$), and violation counts $V_k(x) = \sum_{\ell=1}^r \mathbb{I}\{Y_{k,g,\ell}(x) > 0\}$, merging repeat visits to identical policies via exact parallel variance formulas.

\subsection{Four-Pillar Functional Meta-Prior and Scope of Offline Learning}

Given an offline legacy archive $\mathcal{D}_{\mathrm{src}} = \{(m_a, x_a, r_a, \bar{Y}_{f,a}, S_{f,a}^2, \bar{Y}_{g,a}, S_{g,a}^2, \Xi_{m_a}(x_a))\}_{a=1}^{N_{\mathrm{src}}}$ logged across $M_{\mathrm{src}}$ source tasks by adaptive legacy solvers, offline meta-prior extraction operationalizes four structural inductive biases—\emph{occupancy-balanced orthogonality}, \emph{manifold smoothness}, \emph{low-order additive/interaction coupling}, and \emph{sparse cross-task invariance}—within a prescribed feature dictionary:
\begin{enumerate}
  \item \textbf{Empirical Inverse-Occupancy Balancing.} Standardizing $\tilde{\psi}_a = \operatorname{diag}(\hat{\sigma}_\psi)^{-1}(\psi_a - \hat{\mu}_\psi)$ and forming Gaussian kernel affinities $W_{ab} = \exp(-\|\tilde{\psi}_a - \tilde{\psi}_b\|_2^2 / (2 h_{\mathrm{occ}}^2))$ with empirical kernel occupancy $\hat{\pi}_a = N_{\mathrm{src}}^{-1}\sum_{b=1}^{N_{\mathrm{src}}} W_{ab}$, we compute self-normalized clipped weights ($\bar{w} = 6$)
  \begin{equation}\label{eq:inv_occ_weights}
    w_a = \frac{\min\{\bar{w},\; \operatorname{med}(\hat{\pi}^{-1})^{-1}\hat{\pi}_a^{-1}\}}{\sum_{b=1}^{N_{\mathrm{src}}} \min\{\bar{w},\; \operatorname{med}(\hat{\pi}^{-1})^{-1}\hat{\pi}_b^{-1}\}},
    \qquad \hat{\nu}_w = \sum_{a=1}^{N_{\mathrm{src}}} w_a \delta_{\tilde{\psi}_a}.
  \end{equation}
  This heuristic kernel occupancy balancing reduces dense solver-cluster dominance on observed archive support, though it does not recover unvisited regions of $\mathcal{M}_\psi$ or eliminate finite-sample density-estimation error.

  \item \textbf{Prescribed Low-Order Dictionary and Graph-Laplacian Low-Pass Filtering.} Standardized exposure coordinates are first clipped componentwise to $\bar{\psi}_j = \operatorname{clip}(\tilde{\psi}_j, -4, 4)$ ($0.12\%$--$0.59\%$ of coordinates exceed $\pm 4$ across the $468$-record source archives, and $2.5\%$--$2.8\%$ on target candidate pools). Over $\bar{\psi} \in [-4,4]^{\PsiDim}$, a fixed $\RawDictDim$-feature dictionary $\varphi(\bar{\psi})$ combines univariate linear terms $\bar{\psi}_j$, normalized second-order Hermite polynomials $(\bar{\psi}_j^2 - 1)/\sqrt{2}$, bounded trigonometric modes $\sin(\frac{\pi}{2}\operatorname{clip}(\bar{\psi}_j/2.5, -1, 1))$, and pre-specified action--regime cross-interactions $\bar{\psi}_j \bar{\psi}_k$ (so learned functional couplings operate within this $\PsiDim$-summary dictionary rather than discovering arbitrary raw-coordinate interactions). Using the normalized graph Laplacian $L_{\mathrm{sym}} = I - \Lambda_{\mathrm{deg}}^{-1/2} W \Lambda_{\mathrm{deg}}^{-1/2}$ with $\Lambda_{\mathrm{deg}} = \operatorname{diag}(W\mathbf{1})$, features above the $0.75$ quantile of $\psi$-graph Dirichlet roughness $\bar{\varphi}_j^\top L_{\mathrm{sym}} \bar{\varphi}_j / (\|\bar{\varphi}_j\|_2^2 + \epsilon_0)$ are removed ($39$ smooth dictionary features survive; smoothness on the $\psi$-similarity graph is distinct from temporal low-frequency DCT smoothness in policy coordinates).

  \item \textbf{Empirical-Source Gram Whitening, Sparse Mode Selection, and Two-Stage Covariance Shrinkage.} Centering $\mathring{\varphi} = \varphi_{\mathrm{lp}} - \E_{\hat{\nu}_w}[\varphi_{\mathrm{lp}}]$ and eigendecomposing $G_w = \sum_a w_a \mathring{\varphi}_a \mathring{\varphi}_a^\top = U\Sigma_{\mathrm{gram}} U^\top$, truncated spectral whitening $Q_{\mathrm{orth}} = \Sigma_{\mathrm{gram},+}^{-1/2}U_+^\top$ ($\Sigma_{\mathrm{gram},k} > \max\{10^{-5}\Sigma_{\mathrm{gram},\max}, 10^{-8}\}$) gives the unclipped archive projections $\varphi_a^{\mathrm{orth}} = Q_{\mathrm{orth}}\mathring{\varphi}_a$ unit Gram matrix under $\hat{\nu}_w$. Source per-task ridge-GLS fits $(\hat{\theta}_{m,f}, \hat{\theta}_{m,g})$ weighted by $w_a r_a$ (incorporating both $r_a=1$ and $r_a \ge 2$ logs) regress directly on unclipped $\varphi_a^{\mathrm{orth}}$ ($0.007\%$--$0.061\%$ of source archive coordinates exceed $\pm 5$), whereas downstream feature evaluation clips whitened coordinates componentwise to $\tilde{\varphi}(\psi) = \operatorname{clip}(Q_{\mathrm{orth}}\mathring{\varphi}(\psi), -5, 5)$ to bound extrapolation on out-of-support target queries ($1.1\%$--$1.3\%$ of coordinates in traffic candidate pools and $6.7\%$--$16.8\%$ in queueing pools; both clipping and query-distribution shift alter the Gram matrix away from identity). Modes are ranked by cross-task signal-to-dispersion $\sum_{c \in \{f,g\}} \omega_c \big(\bar{\theta}_{k,c}^2 / (\Var_m(\hat{\theta}_{m,k,c}) + \epsilon_0) + \tfrac{1}{2}\E_m[\hat{\theta}_{m,k,c}^2]\big)$ ($\omega_f = 1, \omega_g = 1.5$), selecting a design-fixed active rank $K_{\mathrm{act}} = \ActiveRank$ ($\tilde{\phi}(x) = [1, \tilde{\varphi}_{S_{\mathrm{act}}}(\psi(x))^\top]^\top \in \R^{p}$, $p = 1 + K_{\mathrm{act}} = 9$). Setting $\mu_{0,c} = \bar{\theta}_{S_{\mathrm{act}},c}$ and empirical sample covariance $\Sigma_{0,c}^{\mathrm{emp}} = \widehat{\operatorname{Cov}}_m(\hat{\theta}_{m,S_{\mathrm{act}},c})$, offline prior fitting applies finite-source meta-mean inflation and baseline diagonal stabilization ($s_0 = 0.15$, $D_{0,c} = \operatorname{diag}(\max\{\operatorname{diag}(\Sigma_{0,c}^{\mathrm{emp}}), 10^{-3}\})$), followed online in \texttt{tempered\_sure\_release} by dimension-aware shrinkage $\omega_{\mathrm{src}} = p / (M_{\mathrm{src}} + p)$ ($= 0.60$ for $M_{\mathrm{src}} = 6, p = 9$):
  \begin{equation}\label{eq:finite_source_cov}
    \Sigma_{0,c} = \Big(1 + \frac{1}{M_{\mathrm{src}}}\Big)\big((1 - s_0)\Sigma_{0,c}^{\mathrm{emp}} + s_0 D_{0,c}\big) + 10^{-4} I_p,
    \qquad
    \widehat{\Sigma}_{0,c}^{\mathrm{stab}} = (1 - \omega_{\mathrm{src}})\,\Sigma_{0,c} + \omega_{\mathrm{src}}\,\operatorname{diag}(\Sigma_{0,c}),
  \end{equation}
  mitigating rank-$\min\{M_{\mathrm{src}}-1, p\}$ sample-covariance instability across small source cohorts. Pruned modes $k \in S_{\mathrm{act}}^c$ form energy-weighted GP covariance features $w_{\perp,c}(x) = \tfrac{1}{2}\,\tilde{\varphi}_{S_{\mathrm{act}}^c}(\psi(x)) \odot \sqrt{\hat{e}_{\perp,c}}$ so pruned-subspace uncertainty contracts at observed target policies.

  \item \textbf{Multi-Replication HVD Hyperprior and Channel-Preserving DCT Templates.} Over the $14$-feature nonnegative basis $u_{\mathrm{HVD}}(x) = [1,\, A_{\mathrm{local}}(x)^{\odot 2},\, N_{\mathrm{shared}}(x)^{\odot 2},\, \{\tfrac{1}{2}(N_i(x)+N_j(x))^2\}_{1 \le i < j},\, N_{\mathrm{shared}}(x)]^\top \in \R_+^{14}$, degrees-of-freedom-weighted NNLS fits source variance weights $\gamma_{0,c}$ \emph{strictly} on multi-replication records $\mathcal{A}_{\mathrm{rep}} = \{a : r_a \ge 2\}$, never mistaking $S_{c,a}^2 = 0$ on $r_a = 1$ logs for zero variance. Elite feasible source policies—screened using HVD-imputed dispersion whenever $r_a = 1$—are compressed per control channel ($C_k \in \{1, 2\}$) via orthonormal DCT-II into $K_{\mathrm{dct}} = 12$ temporal/spatial modes, providing optional horizon-invariant warm-start templates alongside the functional meta-prior $(\tilde{\phi}, \mu_{0,c}, \widehat{\Sigma}_{0,c}^{\mathrm{stab}})$.
\end{enumerate}
