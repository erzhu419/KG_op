\section{Theoretical Analysis and Approximation Scope}\label{sec:theory}

We establish four supporting structural results and two boundary propositions—covering self-normalized occupancy-weight clipping, shift-invariant within-policy dispersion and scalar transfer dominance, exact Gaussian Stein dominance alongside distribution-free excess-risk boundedness in a fixed-design shrinkage reference model, and shortlist-menu decomposition, vanishing-innovation limits, screened truncation boundaries, and distribution-free sequential clean-up safety—and delineate explicitly how each idealized model relates to the online algorithm. Mechanized proof audits under Lean~4.31.0 appear in Electronic Companion Section~\ref*{ec:lean_map}.

\subsection{Reference Occupancy Clipping, Shift Invariance, and Scalar Transfer Boundary}

Let $\nu$ denote a target reference probability measure on exposure domain $\mathcal{M}_\psi$ and $\pi_{\mathrm{occ}} > 0$ an oracle occupancy distribution with density ratio $\rho(\psi) = \nu(\psi)/\pi_{\mathrm{occ}}(\psi)$.

\begin{proposition}[Reference Self-Normalized Clipping Bias and Parseval Separation]\label{prop:subspace}
Let clipped oracle ratios $\tilde{w}(\psi) = \min\{\bar{w}, \rho(\psi)\}$ define $\nu_{\bar{w}}(\mathrm{d}\psi) = \tilde{w}(\psi)\pi_{\mathrm{occ}}(\mathrm{d}\psi)$ and the self-normalized measure $\tilde{\nu}_{\bar{w}}(\mathrm{d}\psi) = \nu_{\bar{w}}(\mathrm{d}\psi)/Z_{\bar{w}}$ with $Z_{\bar{w}} = \int_{\mathcal{M}_\psi} \nu_{\bar{w}}(\mathrm{d}\psi) > 0$. Then:
\begin{enumerate}
  \item \emph{(Unnormalized and Self-Normalized Clipping Bias)} For any $f, h \in L^2(\nu)$, $|\langle f, h \rangle_{\nu_{\bar{w}}} - \langle f, h \rangle_\nu| \le \int_{\mathcal{M}_\psi} \max\{0, \rho(\psi) - \bar{w}\}|f(\psi)h(\psi)|\pi_{\mathrm{occ}}(\mathrm{d}\psi)$. Whenever $|f(\psi)h(\psi)| \le B$ almost everywhere, $Z_{\bar{w}}$ cancels algebraically to yield
  \begin{equation}\label{eq:norm_clip_bias}
    \big|\langle f, h \rangle_{\tilde{\nu}_{\bar{w}}} - \langle f, h \rangle_\nu\big| \;\le\; 2 B \int_{\mathcal{M}_\psi} \max\{0,\, \rho(\psi) - \bar{w}\}\,\pi_{\mathrm{occ}}(\mathrm{d}\psi),
  \end{equation}
  vanishing identically whenever $\sup_\psi \rho(\psi) \le \bar{w}$.
  \item \emph{(Active--Pruned Orthonormal Parseval Separation)} For any $\tilde{\nu}_{\bar{w}}$-orthonormal dictionary $\{\tilde{\varphi}_k\}_{k=1}^{M_+}$ partitioned into $S_{\mathrm{act}}$ and $S_{\mathrm{act}}^c$, any $h(x) = \theta_0 + \sum_{k=1}^{M_+} \theta_k \tilde{\varphi}_k(\psi(x)) + r_{\mathrm{hf}}(x)$ with $r_{\mathrm{hf}} \perp \operatorname{span}(1, \tilde{\varphi})$ satisfies $\|h - \theta_0\|_{L^2(\tilde{\nu}_{\bar{w}})}^2 = \sum_{j \in S_{\mathrm{act}}} \theta_j^2 + \sum_{k \in S_{\mathrm{act}}^c} \theta_k^2 + \|r_{\mathrm{hf}}\|_{L^2(\tilde{\nu}_{\bar{w}})}^2$.
\end{enumerate}
\end{proposition}

Proposition~\ref{prop:subspace} quantifies the clipping-induced inner-product bias given exact density ratios $\rho(\psi)$ together with orthogonal active--pruned energy separation; in implementation~\eqref{eq:inv_occ_weights}, $\rho(\psi)$ is approximated on observed archive support via inverse kernel occupancy $\hat{\pi}_a^{-1}$, so additional finite-sample KDE error and unvisited-support gaps remain.

\begin{theorem}[Within-Policy Shift Invariance and Scalar Transfer Dominance Boundary]\label{thm:transfer_boundary}
Consider evaluating policy $x$ on target task $t$ where $Y_{t,g,\ell}(x) = \mu_{0,g}(x) + \Delta_g(x) + \varepsilon_\ell$ ($\ell = 1,\dots,r$) with independent zero-mean errors $\E[\varepsilon_\ell]=0$ and $\E[\varepsilon_\ell^2] = v_t(x)$.
\begin{enumerate}
  \item \emph{(Mean-Shift Contamination vs.\ Within-Policy Invariance)} Raw prior residuals satisfy $\E[(Y_{t,g,\ell}(x) - \mu_{0,g}(x))^2] = \Delta_g(x)^2 + v_t(x)$, whereas $S_{t,g}^2(x) = \sum_{\ell=1}^r (Y_{t,g,\ell}(x) - \bar{Y}_{t,g}(x))^2 = \sum_{\ell=1}^r (\varepsilon_\ell - \bar{\varepsilon})^2$ cancels $\mu_{0,g}(x) + \Delta_g(x)$ deterministically on every sample path and satisfies $\E[S_{t,g}^2(x)/(r-1)] = v_t(x)$ for any fixed batch size $r \ge 2$.
  \item \emph{(Finite-$M_{\mathrm{src}}$ Mismatch Decomposition)} For a scalar coordinate where $\hat{\mu}_0 = \mu^\star + \xi_{\mathrm{meta}} + b_{\mathrm{occ}}$ ($\E[\xi_{\mathrm{meta}}]=0$, $\Var(\xi_{\mathrm{meta}}) = \sigma_{\mathrm{src}}^2 / M_{\mathrm{src}}$) and $\theta_t = \mu^\star + \Delta_{\mathrm{shift}}$, $\Delta_{\mathrm{eff}}^2 \coloneqq (\theta_t - \hat{\mu}_0)^2$ satisfies $\E[\Delta_{\mathrm{eff}}^2] = (\Delta_{\mathrm{shift}} - b_{\mathrm{occ}})^2 + \sigma_{\mathrm{src}}^2 / M_{\mathrm{src}}$.
  \item \emph{(Scalar Linear Transfer Boundary)} Conditioning on $\hat{\mu}_0$ and combining prior precision $\kappa_0 > 0$ with an unbiased target estimator of precision $\rho_n > 0$ yields $\operatorname{MSE}(\kappa_0; \Delta_{\mathrm{eff}}) = (\kappa_0^2\Delta_{\mathrm{eff}}^2 + \rho_n)/(\kappa_0 + \rho_n)^2$, which strictly beats target-only estimation ($\rho_n^{-1}$) \textbf{if and only if}
  \begin{equation}\label{eq:transfer_iff}
    \operatorname{MSE}(\kappa_0; \Delta_{\mathrm{eff}}) < \frac{1}{\rho_n}
    \qquad \Longleftrightarrow \qquad
    \Delta_{\mathrm{eff}}^2 < \frac{1}{\rho_n} + \frac{2}{\kappa_0}.
  \end{equation}
\end{enumerate}
\end{theorem}

While deterministic mean cancellation holds on any pooled sample, adaptive stopping or data-dependent revisiting can bias pooled sample variances across sequential batches (e.g., two symmetric $\{-1,1\}$ draws followed by two extra draws only upon initial disagreement yield $\E[\hat{s}_{\mathrm{pool}}^2] = 7/12 < 1$; Section~\ref*{ec:proofs}). We therefore audit both fixed-batch and pooled variance MSEs empirically on matched target paths.

\subsection{Reference Gaussian Stein Dominance and Distribution-Free \texttt{SURE} Risk Cap}

Theorem~\ref{thm:transfer_boundary}(3) motivates shrinking or releasing source priors when $\Delta_{\mathrm{eff}}^2$ is large. Consider an \emph{idealized fixed-design, known-precision linear shrinkage reference model} over an $m_{\mathrm{id}}$-dimensional identifiable subspace $\mathcal{I}$ ($m_{\mathrm{id}} \ge 1$) where whitened projections $\tilde{R}_k = \delta_k + \xi_k$ ($k \in \mathcal{I}$) combine unknown prior--target discrepancies $\delta_k$ with zero-mean errors $\xi_k$, comparing linear shrinkage $\hat{\theta}_k(\lambda) = (1-\lambda)\mu_{0,k} + \lambda(\mu_{0,k} + \tilde{R}_k)$ against target-only estimation ($\lambda = 1$) via normalized loss $L_{\mathcal{I}}(\lambda) = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} ((1 - \lambda)\delta_k - \lambda \xi_k)^2$ and Stein--Mallows risk proxy $\widehat{R}_{\mathcal{I}}(\lambda) = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \big[(1 - \lambda)^2 \tilde{R}_k^2 + (2\lambda - 1)\sigma_k^2\big]$.

\begin{theorem}[Exact Gaussian Stein Dominance at $m_{\mathrm{id}} \ge 5$ and Non-Gaussian Excess-Risk Cap]\label{thm:sure_uniform}
Consider the reference model above, unclipped Stein--Mallows weight $\tilde{\lambda}_{m_{\mathrm{id}}} = 1 - m_{\mathrm{id}} / \|\tilde{R}\|_2^2$, and positive-part clipped selector $\hat{\lambda}^{\mathrm{SURE}} = \operatorname{clip}_{[0,1]}(1 - m_{\mathrm{id}} / \|\tilde{R}\|_2^2)$ matching~\eqref{eq:sure_closed_form}.
\begin{enumerate}
  \item \emph{(Exact Gaussian Minimax Dominance for $m_{\mathrm{id}} \ge 5$)} Suppose $\bm{\xi} \sim \mathcal{N}(0, I_{m_{\mathrm{id}}})$ ($\sigma_k^2 \equiv 1$, $\E[L_{\mathcal{I}}(1)] = 1$). For every $m_{\mathrm{id}} \ge 3$ and any $c \in \R$, Stein's identity gives $\E[L_{\mathcal{I}}(1 - c/\|\tilde{R}\|_2^2)] = 1 - \frac{2c(m_{\mathrm{id}}-2) - c^2}{m_{\mathrm{id}}}\E[\|\tilde{R}\|_2^{-2}]$. At $c = m_{\mathrm{id}}$, $2 m_{\mathrm{id}}(m_{\mathrm{id}} - 2) - m_{\mathrm{id}}^2 = m_{\mathrm{id}}(m_{\mathrm{id}} - 4)$, so for every $m_{\mathrm{id}} \ge 5$ and every $\bm{\delta} \in \R^{m_{\mathrm{id}}}$, both $\tilde{\lambda}_{m_{\mathrm{id}}}$ and positive-part $\hat{\lambda}^{\mathrm{SURE}}$ \textbf{strictly dominate} target-only estimation:
  \begin{equation}\label{eq:stein_dominance}
    \E\big[L_{\mathcal{I}}(\hat{\lambda}^{\mathrm{SURE}})\big]
    \;<\;
    \E\big[L_{\mathcal{I}}(\tilde{\lambda}_{m_{\mathrm{id}}})\big]
    \;=\;
    1 - (m_{\mathrm{id}} - 4)\,\E\!\left[\frac{1}{\|\tilde{R}\|_2^2}\right]
    \;<\; 1
    \qquad (\forall\, m_{\mathrm{id}} \ge 5,\; \forall\, \bm{\delta} \in \R^{m_{\mathrm{id}}}).
  \end{equation}
  At $m_{\mathrm{id}} = 4$, $\E[L_{\mathcal{I}}(\tilde{\lambda}_4)] = 1$ and positive-part clipping still yields $\E[L_{\mathcal{I}}(\hat{\lambda}^{\mathrm{SURE}})] < 1$.
  \item \emph{(Distribution-Free Sample-Path Identity and Finite-Moment Cap for All $m_{\mathrm{id}} \ge 1$)} Without Gaussianity, let $\hat{\lambda} \in [0, 1]$ be any selector satisfying $\widehat{R}_{\mathcal{I}}(\hat{\lambda}) \le \widehat{R}_{\mathcal{I}}(1)$ (including $\hat{\lambda}^{\mathrm{SURE}}$ at $\sigma_k^2 \equiv 1$), and write $\bar{\delta}^2 = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \delta_k^2$, $\bar{C} = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \delta_k \xi_k$, and $\bar{\sigma}^2 = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \sigma_k^2$. Setting $u = 1 - \lambda$, $\big(L_{\mathcal{I}}(\lambda) - L_{\mathcal{I}}(1)\big) - \frac{3}{2}\big(\widehat{R}_{\mathcal{I}}(\lambda) - \widehat{R}_{\mathcal{I}}(1)\big) = -\frac{1}{2}u^2 \bar{\delta}^2 - u(2 + u)\bar{C} - (2u + \frac{1}{2}u^2)\frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \xi_k^2 + 3u\bar{\sigma}^2$, which implies $L_{\mathcal{I}}(\hat{\lambda}) - L_{\mathcal{I}}(1) \le 3\bar{\sigma}^2$ at $\bar{\delta}^2 = 0$ and
  \begin{equation}\label{eq:sure_uniform_path}
    L_{\mathcal{I}}(\hat{\lambda}) - L_{\mathcal{I}}(1) \;\le\; 3\bar{\sigma}^2 + \frac{9\bar{C}^2}{2\bar{\delta}^2}
    \qquad (\forall\, \bm{\delta} \neq 0).
  \end{equation}
  Whenever $\{\xi_k\}_{k \in \mathcal{I}}$ are pairwise orthogonal with $\E[\xi_k^2] \le \sigma_{\max}^2$ and $\sigma_k^2 \le \sigma_{\max}^2$, $\sup_{\bm{\delta} \in \R^{m_{\mathrm{id}}}} \E[L_{\mathcal{I}}(\hat{\lambda}) - L_{\mathcal{I}}(1)] \le (3 + \frac{9}{2 m_{\mathrm{id}}})\sigma_{\max}^2$.
\end{enumerate}
\end{theorem}

\paragraph{Interpretation and relationship to the online GP implementation.}
Parts~1 and~2 play complementary roles. Under Gaussian whitened errors,~\eqref{eq:stein_dominance} shows that choosing $c = m_{\mathrm{id}}$ via Mallows--Stein unbiased risk estimation achieves strict minimax dominance over target-only estimation ($\lambda = 1$) across all $\bm{\delta} \in \R^{m_{\mathrm{id}}}$ once $m_{\mathrm{id}} \ge 5$ (relevant across our $K_{\mathrm{act}} = 8$-mode active basis once initial policies span at least five directions; \citealp{james1961estimation,baranchik1970family}). Conversely, during very early low-rank stages ($1 \le m_{\mathrm{id}} \le 3$, where $c = m_{\mathrm{id}}$ exceeds the Gaussian James--Stein range $2(m_{\mathrm{id}}-2)$) or under skewed non-Gaussian simulation batch errors where Stein's Gaussian lemma fails, Part~2 guarantees that excess risk remains uniformly bounded over arbitrary shifts $\bm{\delta}$ rather than growing as $\propto \bar{\delta}^2$. Even so, Theorem~\ref{thm:sure_uniform} remains a \emph{design-motivating reference result} rather than an end-to-end bound for online \textbf{GP-FTA-ECKG}, because: (i)~plug-in HVD precision weights $\hat{W}_{n,c}$ differ from exact $V_{n,c}^{-1}$; (ii)~sequential queries are chosen adaptively \citep{deshpande2018accurate}; (iii)~$\hat{\lambda}_{n,c}^{\mathrm{SURE}}$ enters composite GP moments~\eqref{eq:sure_mean}--\eqref{eq:sure_kernel}; and (iv)~subspace risk on $\mathcal{I}_{n,c}$ does not bound unidentifiable directions or $\psi$-compression error. Finite-sample deployment safety therefore rests strictly on Theorem~\ref{thm:vkg_safety}.

\subsection{Shortlist-Menu Decomposition, Consistency Boundaries, and Clean-Up Safety}\label{sec:cascade_value}

\begin{theorem}[Shortlist-Menu \texttt{EC-KG} Decomposition and Single-Slot Innovation Cap]\label{thm:eckg_dual_voi}\label{thm:sequential_bgpkg}
Fix search state $\mathcal{F}_n$ with evaluated policy set $\mathcal{D}_n$, non-negative fantasy weights $\{w_s\}$ ($\sum_s w_s = 1$), working-model cascade utility $U_{n+1}(\mathcal{S}; q, \xi^{(s)})$ ($U_{n+1}(\varnothing) \equiv 0$), current reference shortlist $\mathcal{S}_n^\star$, and nested shortlist menus $\mathfrak{M}_0^{(s)} = \{\varnothing, \mathcal{S}_n^\star, \hat{\mathcal{S}}_{n+1}(\mathcal{D}_n)\} \subseteq \mathfrak{M}_1^{(s)}(q) = \mathfrak{M}_0^{(s)} \cup \{\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n \cup \{q\})\}$. Then:
\begin{enumerate}
  \item \emph{(Menu-Proxy Dual Decomposition)} Generalizing sampled-design \texttt{KGCP} \citep{scott2011correlated} to shortlist menus, every $q \in \mathcal{C}_n$ satisfies $\widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q) = \widehat{\mathrm{VOI}}_n^{\mathrm{refine}}(q) + \widehat{\mathrm{VOI}}_n^{\mathrm{admit}}(q)$ with $\widehat{\mathrm{VOI}}_n^{\mathrm{refine}}(q) \ge 0$, $\widehat{\mathrm{VOI}}_n^{\mathrm{admit}}(q) \ge 0$, and $\widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q) \ge 0$.
  \item \emph{(Revisit Zero-Admission Identity)} For every evaluated revisit $q \in \mathcal{D}_n$, $\mathcal{D}_n \cup \{q\} = \mathcal{D}_n$ implies $\widehat{\mathrm{VOI}}_n^{\mathrm{admit}}(q) = 0$ identically, so $\widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q) = \widehat{\mathrm{VOI}}_n^{\mathrm{refine}}(q)$.
  \item \emph{(Single-Slot Submartingale and Lipschitz Cap)} On a factored single-slot evaluation set $\mathcal{C}_n^{\mathrm{eval}}$ where $\E_n[\mu_{n+1,f}(x')] = \mu_{n,f}(x')$, $\E_n[p_{n+1,0}^{\mathrm{gate}}(x')] = p_{n,0}^{\mathrm{gate}}(x') \in [0,1]$, $|\mu_{n+1,f}(x')-\mu_{n,f}(x')| \le \varepsilon_f(q)$, $|p_{n+1,0}^{\mathrm{gate}}(x')-p_{n,0}^{\mathrm{gate}}(x')| \le \varepsilon_g(q)$, and $(\hat{J}_n^{\mathrm{ref}} - \mu_{n,f}(x'))_+ \le B_f$, single-slot utility $V_{n+1}(x')$ satisfies $0 \le \E_n[\max_{x' \in \mathcal{C}_n^{\mathrm{eval}}} V_{n+1}(x')] - \max_{x' \in \mathcal{C}_n^{\mathrm{eval}}} V_n(x') \le \varepsilon_f(q) + B_f\varepsilon_g(q)$.
\end{enumerate}
\end{theorem}

\begin{proposition}[Vanishing-Innovation Property vs.\ Screened Boundary]\label{prop:consistency_boundary}
Consider a finite candidate set $\mathcal{C}$ with positive observation variances:
\begin{enumerate}
  \item \emph{(Unscreened No-Starvation Property)} Whenever $q \in \mathcal{D}_n$ satisfies $|U_{n+1}(\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n); q, \xi^{(s)}) - U_{n+1}(\mathcal{S}_n^\star; q, \xi^{(s)})| \le \varepsilon_n(q)$ across fantasy nodes (with $U_{n+1}(\mathcal{S}_n^\star) \ge 0$), $\widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q) \le \varepsilon_n(q)$. Consequently, along any sample path where revisiting $q$ infinitely often drives one-step posterior innovations $\varepsilon_n(q) \to 0$, unscreened \textbf{EC-KG} ($\mathcal{K}_n = \mathcal{C}$) cannot sample only policies in a subset $\mathcal{A} \subset \mathcal{C}$ infinitely often while any $q' \in \mathcal{C} \setminus \mathcal{A}$ retains $\liminf_{n \to \infty} \widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q') > 0$.
  \item \emph{(Constructive Screened Counterexample)} Under a fixed Stage~A rank-one single-slot contender cap $|\mathcal{K}_n| \le K_{\mathrm{cap}} < |\mathcal{C}|$, there exist search states where a backup contender $q_{\mathrm{B}}$ has zero single-slot marginal score $V_n(q_{\mathrm{B}}) = 0$ (and is deterministically excluded behind $K_{\mathrm{cap}}$ rivals with small positive $V_n > 0$), yet strictly maximizes both $\widehat{\mathrm{VOI}}_n^{\mathrm{EC}}$ and unconditional single-nomination gain $\Delta_n^{\mathrm{term}}$ whenever Slot-0 pass power satisfies $p_0 < 1$.
\end{enumerate}
\end{proposition}

\paragraph{Menu proxy vs.\ single nomination and screening trade-off.}
Proposition~\ref{prop:consistency_boundary} formalizes both the asymptotic behavior of unscreened \textbf{EC-KG} on finite pools and why Stage~A's finite-budget contender cap ($|\mathcal{K}_n| \le 16$) is an intentional computational approximation that can miss late-stage backup-slot winners. Likewise, because terminal clean-up invokes the nominator once to form $\hat{\mathcal{S}}_N(\mathcal{D}_N)$ rather than maximizing over $\{\varnothing, \mathcal{S}_{N-1}^\star, \hat{\mathcal{S}}_N(\mathcal{D}_N)\}$, single-nomination gain $\Delta_n^{\mathrm{term}}(q) = \E_n[U_{n+1}(\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n \cup \{q\})) - U_{n+1}(\hat{\mathcal{S}}_n(\mathcal{D}_n))]$ need not be nonnegative and can rank high-variance shortlist perturbations differently from $\widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q)$, as audited directly in Section~\ref{sec:results}.

\begin{theorem}[Distribution-Free Clean-Up Safety and Exact Sequential Binomial Curtailment]\label{thm:vkg_safety}
Condition on any frozen shortlist $\mathcal{S} = (x_{(0)}, \dots, x_{(L-1)})$ ($1 \le L \le 3$) and search history $\mathcal{F}_N$. Suppose verification constraint indicators $I_{i,1},\dots,I_{i,K_i} \in \{0,1\}$ at each slot $i$ are i.i.d.\ Bernoulli($p_t^{\mathrm{ver}}(x_{(i)})$) under verification law $\mathbb{Q}_t^{\mathrm{ver}}$ (for finite-block historical benchmarks, $\mathbb{Q}_t^{\mathrm{ver}}$ denotes the empirical law of block trajectories drawn with replacement from the designated verification pool), and let $k_i^\star$ satisfy $\operatorname{BinomCDF}(k_i^\star; K_i, \alpha) \le \delta_i$ with $\sum_{i=0}^{L-1} \delta_i \le \delta$.
\begin{enumerate}
  \item \emph{(Exact Familywise False-Certification Bound)} Under both the incumbent-guarded branch and the sequential first-pass branch,
  \begin{equation}\label{eq:fwer_safety}
    \Prob_{\mathbb{Q}_t^{\mathrm{ver}}}\!\big(\mathrm{state} = \texttt{certified\_deployment} \;\text{and}\; p_t^{\mathrm{ver}}(\hat{x}_t) > \alpha \,\big|\, \mathcal{F}_N\big) \;\le\; \delta,
  \end{equation}
  using at most $B_{\mathrm{ver}} = \sum_{i=0}^{L-1} K_i$ replications ($\MaxVerifCalls$ for $(K_0,K_1,K_2) = (\SlotZeroK, \SlotOneK, \SlotTwoK)$, $k^\star = 1$ at $\alpha = \delta = 0.05$).
  \item \emph{(Sample-Path Equivalence of Sequential Binomial Curtailment)} Writing $V_{i,m} = \sum_{\ell=1}^m I_{i,\ell}$, stopping any single-gate slot $i$ at $\tau_i = \min\big\{m \in \{1,\dots,K_i\} : V_{i,m} \ge k_i^\star + 1 \;\text{or}\; V_{i,m} + (K_i - m) \le k_i^\star\big\}$ (completing full blocks $(K_0, K_1)$ only when both guarded slots pass) yields $\{V_{i,\tau_i} \le k_i^\star\} = \{V_{i,K_i} \le k_i^\star\}$ identically on every sample path—preserving the exact certification decision, deployed policy, and bound~\eqref{eq:fwer_safety} while strictly reducing expected verification calls whenever $p_t^{\mathrm{ver}}(x_{(i)}) \in (0, 1)$.
\end{enumerate}
\end{theorem}

\paragraph{Archive Call Amortization.}
When historical trajectories are pre-existing sunk enterprise logs ($B_{\mathrm{src}}^{\mathrm{marginal}} = 0$), any per-task call reduction $\Delta B = B_{\mathrm{cold}} - B_{\mathrm{FTA}} > 0$ at equal or lower deployment loss yields immediate savings ($M_{\mathrm{break}}^{\mathrm{sunk}} = 1$); if $B_{\mathrm{src}} > 0$ offline source calls are generated specifically to build the archive across $M$ target tasks, full simulation cost breaks even at $M \ge M_{\mathrm{break}}^{\mathrm{full}} = \lceil B_{\mathrm{src}} / \Delta B \rceil$ (Table~\ref*{tab:amortization}).
