\section{Theoretical Analysis and Approximation Scope}\label{sec:theory}

We establish four formal results—covering self-normalized occupancy-weight clipping, shift-invariant within-policy dispersion and scalar transfer dominance, uniform excess risk of Stein--Mallows shrinkage in a fixed-design reference model, and shortlist-menu decomposition alongside distribution-free clean-up safety—and delineate explicitly how each theoretical model relates to the online algorithm. Core algebraic and measure-theoretic statements are machine-checked in Lean~4.31.0 (Section~\ref*{ec:lean_map}).

\subsection{Reference Occupancy Clipping, Shift Invariance, and Scalar Transfer Boundary}

Let $\nu$ denote a target reference probability measure on exposure domain $\mathcal{M}_\psi$ and $\pi_{\mathrm{occ}} > 0$ an oracle occupancy distribution with density ratio $\rho(\psi) = \nu(\psi)/\pi_{\mathrm{occ}}(\psi)$.

\begin{proposition}[Reference Self-Normalized Clipping Bias and Parseval Separation]\label{prop:subspace}
Let clipped oracle ratios $\tilde{w}(\psi) = \min\{\bar{w}, \rho(\psi)\}$ define $\nu_{\bar{w}}(\mathrm{d}\psi) = \tilde{w}(\psi)\pi_{\mathrm{occ}}(\mathrm{d}\psi)$ and the self-normalized measure $\tilde{\nu}_{\bar{w}}(\mathrm{d}\psi) = \nu_{\bar{w}}(\mathrm{d}\psi)/Z_{\bar{w}}$ with $Z_{\bar{w}} = \int_{\mathcal{M}_\psi} \nu_{\bar{w}}(\mathrm{d}\psi) > 0$. Then:
\begin{enumerate}
  \item \emph{(Unnormalized and Self-Normalized Clipping Bias)} For any $f, h \in L^2(\nu)$, $|\langle f, h \rangle_{\nu_{\bar{w}}} - \langle f, h \rangle_\nu| \le \int_{\mathcal{M}_\psi} \max\{0, \rho(\psi) - \bar{w}\}|f(\psi)h(\psi)|\pi_{\mathrm{occ}}(\mathrm{d}\psi)$. Whenever $|f(\psi)h(\psi)| \le B$ almost everywhere, $Z_{\bar{w}}$ cancels algebraically to yield
  \begin{equation}\label{eq:norm_clip_bias}
    \big|\langle f, h \rangle_{\tilde{\nu}_{\bar{w}}} - \langle f, h \rangle_\nu\big| \;\le\; 2 B \int_{\mathcal{M}_\psi} \max\{0,\, \rho(\psi) - \bar{w}\}\,\pi_{\mathrm{occ}}(\mathrm{d}\psi),
  \end{equation}
  vanishing identically whenever $\sup_\psi \rho(\psi) \le \bar{w}$.
  \item \emph{(Active--Pruned Orthonormal Parseval Separation)} For any $\tilde{\nu}_{\bar{w}}$-orthonormal dictionary $\{\tilde{\varphi}_k\}_{k=1}^{M_+}$ partitioned into $S_{\mathrm{act}}$ and $S_{\mathrm{act}}^c$, any $h(x) = \theta_0 + \sum_{k=1}^{M_+} \theta_k \tilde{\varphi}_k(\psi(x)) + r_{\mathrm{hf}}(x)$ with $r_{\mathrm{hf}} \perp \operatorname{span}(1, \tilde{\varphi})$ satisfies $\|h - \theta_0\|_{L^2(\tilde{\nu}_{\bar{w}})}^2 = \sum_{j \in S_{\mathrm{act}}} \theta_j^2 + \sum_{k \in S_{\mathrm{act}}^c} \theta_k^2 + \|r_{\mathrm{hf}}\|_{L^2(\tilde{\nu}_{\bar{w}})}^2$.
\end{enumerate}
\end{proposition}

Proposition~\ref{prop:subspace} characterizes the bias--variance trade-off of weight clipping given exact density ratios $\rho(\psi)$; in implementation~\eqref{eq:inv_occ_weights}, $\rho(\psi)$ is approximated on observed archive support via inverse kernel occupancy $\hat{\pi}_a^{-1}$, so additional finite-sample KDE error and unvisited-support gaps remain.

\begin{theorem}[Within-Policy Shift Invariance and Scalar Transfer Dominance Boundary]\label{thm:transfer_boundary}
Consider evaluating policy $x$ on target task $t$ where $Y_{t,g,\ell}(x) = \mu_{0,g}(x) + \Delta_g(x) + \varepsilon_\ell$ ($\ell = 1,\dots,r$) with independent zero-mean errors $\E[\varepsilon_\ell]=0$ and $\E[\varepsilon_\ell^2] = v_t(x)$.
\begin{enumerate}
  \item \emph{(Mean-Shift Contamination vs.\ Within-Policy Invariance)} Raw prior residuals satisfy $\E[(Y_{t,g,\ell}(x) - \mu_{0,g}(x))^2] = \Delta_g(x)^2 + v_t(x)$, whereas $S_{t,g}^2(x) = \sum_{\ell=1}^r (Y_{t,g,\ell}(x) - \bar{Y}_{t,g}(x))^2 = \sum_{\ell=1}^r (\varepsilon_\ell - \bar{\varepsilon})^2$ cancels $\mu_{0,g}(x) + \Delta_g(x)$ deterministically on every sample path and satisfies $\E[S_{t,g}^2(x)/(r-1)] = v_t(x)$ for any fixed batch size $r \ge 2$.
  \item \emph{(Finite-$M_{\mathrm{src}}$ Mismatch Decomposition)} For a scalar coordinate where $\hat{\mu}_0 = \mu^\star + \xi_{\mathrm{meta}} + b_{\mathrm{occ}}$ ($\E[\xi_{\mathrm{meta}}]=0$, $\Var(\xi_{\mathrm{meta}}) = \sigma_{\mathrm{src}}^2 / M_{\mathrm{src}}$) and $\theta_t = \mu^\star + \Delta_{\mathrm{shift}}$, $\Delta_{\mathrm{eff}}^2 \coloneqq (\theta_t - \hat{\mu}_0)^2$ satisfies $\E[\Delta_{\mathrm{eff}}^2] = (\Delta_{\mathrm{shift}} - b_{\mathrm{occ}})^2 + \sigma_{\mathrm{src}}^2 / M_{\mathrm{src}}$.
  \item \emph{(Scalar Linear Transfer Boundary)} Conditioning on $\hat{\mu}_0$ and combining prior precision $\kappa_0 > 0$ with an unbiased target estimator of precision $\rho_n > 0$ yields $\operatorname{MSE}(\kappa_0; \Delta_{\mathrm{eff}}) = (\kappa_0^2\Delta_{\mathrm{eff}}^2 + \rho_n)/(\kappa_0 + \rho_n)^2$, which strictly beats target-only estimation ($\rho_n^{-1}$) \textbf{if and only if}
  \begin{equation}\label{eq:transfer_iff}
    \operatorname{MSE}(\kappa_0; \Delta_{\mathrm{eff}}) < \frac{1}{\rho_n}
    \qquad \Longleftrightarrow \qquad
    \Delta_{\mathrm{eff}}^2 < \frac{1}{\rho_n} + \frac{2}{\kappa_0}.
  \end{equation}
\end{enumerate}
\end{theorem}

While deterministic mean cancellation holds on any pooled sample, adaptive stopping or data-dependent revisiting can bias pooled sample variances across sequential batches (e.g., two symmetric $\{-1,1\}$ draws followed by two extra draws only upon initial disagreement yield $\E[\hat{s}_{\mathrm{pool}}^2] = 7/12 < 1$; Section~\ref*{ec:proofs}). We therefore audit both fixed-batch and pooled variance MSEs empirically on matched target paths.

\subsection{Reference Linear \texttt{SURE} Risk Bound and Scope of the Online GP Approximation}

Theorem~\ref{thm:transfer_boundary}(3) motivates shrinking or releasing source priors when $\Delta_{\mathrm{eff}}^2$ is large. To analyze Stein--Mallows shrinkage rigorously without overclaiming properties of adaptive plug-in GP estimation, we state Theorem~\ref{thm:sure_uniform} for an \emph{idealized fixed-design, known-precision linear shrinkage reference model} over an $m_{\mathrm{id}}$-dimensional identifiable subspace $\mathcal{I}$ ($m_{\mathrm{id}} \ge 1$) where observed projections $\tilde{R}_k = \delta_k + \xi_k$ ($k \in \mathcal{I}$) combine unknown prior--target discrepancies $\delta_k$ with zero-mean errors $\xi_k$, and compare linear shrinkage $\hat{\theta}_k(\lambda) = (1-\lambda)\mu_{0,k} + \lambda(\mu_{0,k} + \tilde{R}_k)$ against target-only estimation ($\lambda = 1$) via normalized loss $L_{\mathcal{I}}(\lambda) = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} ((1 - \lambda)\delta_k - \lambda \xi_k)^2$ and Stein--Mallows risk proxy $\widehat{R}_{\mathcal{I}}(\lambda) = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \big[(1 - \lambda)^2 \tilde{R}_k^2 + (2\lambda - 1)\sigma_k^2\big]$.

\begin{theorem}[Reference Linear \texttt{SURE} Uniform Shift Excess-Risk Bound]\label{thm:sure_uniform}
In the direct linear shrinkage reference model above, let $\hat{\lambda} \in [0, 1]$ be any selector satisfying $\widehat{R}_{\mathcal{I}}(\hat{\lambda}) \le \widehat{R}_{\mathcal{I}}(1)$ (including $\hat{\lambda} = \operatorname{clip}_{[0,1]}(1 - m_{\mathrm{id}}\bar{\sigma}^2 / \sum_{k \in \mathcal{I}} \tilde{R}_k^2)$). Write $\bar{\delta}^2 = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \delta_k^2$, $\bar{C} = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \delta_k \xi_k$, and $\bar{\sigma}^2 = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \sigma_k^2$. Then:
\begin{enumerate}
  \item \emph{(Algebraic Quadratic Identity)} Setting $u = 1 - \lambda$, for all $\lambda \in \R$,
  \begin{equation}\label{eq:sure_identity}
    \big(L_{\mathcal{I}}(\lambda) - L_{\mathcal{I}}(1)\big) - \frac{3}{2}\big(\widehat{R}_{\mathcal{I}}(\lambda) - \widehat{R}_{\mathcal{I}}(1)\big)
    = -\frac{1}{2}u^2 \bar{\delta}^2 - u(2 + u)\bar{C} - \Big(2u + \frac{1}{2}u^2\Big)\frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \xi_k^2 + 3u\bar{\sigma}^2.
  \end{equation}
  \item \emph{(Sample-Path Excess-Loss Bound)} If $\bar{\delta}^2 = 0$, $L_{\mathcal{I}}(\hat{\lambda}) - L_{\mathcal{I}}(1) \le 3\bar{\sigma}^2$; for every nonzero shift $\bm{\delta} \neq 0$,
  \begin{equation}\label{eq:sure_uniform_path}
    L_{\mathcal{I}}(\hat{\lambda}) - L_{\mathcal{I}}(1) \;\le\; 3\bar{\sigma}^2 + \frac{9\bar{C}^2}{2\bar{\delta}^2},
  \end{equation}
  specializing at unit proxy $\sigma_k^2 \equiv 1$ to $3 + 9\bar{C}^2/(2\bar{\delta}^2)$ (\texttt{fisher\_whitened\_unit\_variance\_sure\_bound}).
  \item \emph{(Conditional Expected Excess Risk Under Known Precision)} Under a fixed design $X_{n,c}$ and known true error covariance $V_{n,c} = W_{n,c}^{-1}$ (so $\operatorname{Cov}(\tilde{\bm{\xi}}) = I_{m_{\mathrm{id}}}$ and $\sigma_k^2 = \E[\xi_k^2] = 1$), $\sup_{\bm{\delta} \in \R^{m_{\mathrm{id}}}} \E[L_{\mathcal{I}}(\hat{\lambda}) - L_{\mathcal{I}}(1)] \le 3 + \frac{9}{2 m_{\mathrm{id}}}$. More generally, if $\{\xi_k\}_{k \in \mathcal{I}}$ are pairwise orthogonal with $\E[\xi_k^2] \le \sigma_{\max}^2$ and $\sigma_k^2 \le \sigma_{\max}^2$, the bound scales by $\sigma_{\max}^2$.
\end{enumerate}
\end{theorem}

\paragraph{Interpretation and relationship to the online GP implementation.}
Theorem~\ref{thm:sure_uniform} serves as a design-motivating reference property: whereas a fixed source prior suffers unbounded estimation loss $\propto \bar{\delta}^2$ as prior--target mismatch grows, any empirical Stein--Mallows selector satisfying $\widehat{R}_{\mathcal{I}}(\hat{\lambda}) \le \widehat{R}_{\mathcal{I}}(1)$ caps expected excess risk uniformly over all $\bm{\delta} \in \R^{m_{\mathrm{id}}}$. Notice, however, that at unit noise variance and rank $m_{\mathrm{id}} = 8$, the worst-case reference bound $3 + 9/(2 m_{\mathrm{id}}) = 3.5625$ corresponds to total normalized risk $\le 4.5625$ versus $1$ for target-only estimation ($\lambda = 1$); thus its mathematical contribution is certifying shift-independent boundedness rather than tight finite-rank dominance. Furthermore, Theorem~\ref{thm:sure_uniform} does \emph{not} directly bound the prediction error or deployment loss of the online \textbf{GP-FTA-ECKG} surrogate for four reasons: (i)~online precision weights $\hat{W}_{n,c} = \operatorname{diag}(r_i / \hat{v}_{n,c}(x_i))$ plug in estimated HVD variances, so the projected error covariance is no longer guaranteed to equal $I_{m_{\mathrm{id}}}$ even on a fixed design; (ii)~sequential query locations and replication counts depend adaptively on past observations \citep{deshpande2018accurate}; (iii)~$\hat{\lambda}_{n,c}^{\mathrm{SURE}}$ enters a composite GP mean and covariance~\eqref{eq:sure_mean}--\eqref{eq:sure_kernel} followed by marginal-likelihood hyperparameter selection rather than direct linear estimation; and (iv)~subspace estimation risk on $\mathcal{I}_{n,c}$ does not control unidentifiable basis directions or $\psi$-representation error. Accordingly, online Fisher-subspace \texttt{SURE} operates as a fast, reference-motivated empirical shrinkage rule, while finite-sample deployment safety rests strictly on Theorem~\ref{thm:vkg_safety}.

\subsection{Shortlist-Menu Proxy Decomposition and Distribution-Free Clean-Up Safety}\label{sec:cascade_value}

\begin{theorem}[Shortlist-Menu \texttt{EC-KG} Decomposition and Single-Slot Innovation Cap]\label{thm:eckg_dual_voi}\label{thm:sequential_bgpkg}
Fix search state $\mathcal{F}_n$ with evaluated policy set $\mathcal{D}_n$, non-negative fantasy weights $\{w_s\}$ ($\sum_s w_s = 1$), working-model cascade utility $U_{n+1}(\mathcal{S}; q, \xi^{(s)})$ ($U_{n+1}(\varnothing) \equiv 0$), current reference shortlist $\mathcal{S}_n^\star$, and nested shortlist menus $\mathfrak{M}_0^{(s)} = \{\varnothing, \mathcal{S}_n^\star, \hat{\mathcal{S}}_{n+1}(\mathcal{D}_n)\} \subseteq \mathfrak{M}_1^{(s)}(q) = \mathfrak{M}_0^{(s)} \cup \{\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n \cup \{q\})\}$. Then:
\begin{enumerate}
  \item \emph{(Menu-Proxy Dual Decomposition)} For every candidate query $q \in \mathcal{C}_n$, the menu-maximization score~\eqref{eq:eckg_score} decomposes as $\widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q) = \widehat{\mathrm{VOI}}_n^{\mathrm{refine}}(q) + \widehat{\mathrm{VOI}}_n^{\mathrm{admit}}(q)$ with $\widehat{\mathrm{VOI}}_n^{\mathrm{refine}}(q) \ge 0$, $\widehat{\mathrm{VOI}}_n^{\mathrm{admit}}(q) \ge 0$, and $\widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q) \ge 0$.
  \item \emph{(Revisit Zero-Admission Identity)} For every already-evaluated revisit $q \in \mathcal{D}_n$, $\mathcal{D}_n \cup \{q\} = \mathcal{D}_n$ implies $\widehat{\mathrm{VOI}}_n^{\mathrm{admit}}(q) = 0$ identically, so $\widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q) = \widehat{\mathrm{VOI}}_n^{\mathrm{refine}}(q)$ (vanishing at zero posterior innovation).
  \item \emph{(Single-Slot Submartingale and Lipschitz Cap)} On a factored single-slot evaluation set $\mathcal{C}_n^{\mathrm{eval}}$ where $\E_n[\mu_{n+1,f}(x')] = \mu_{n,f}(x')$, $\E_n[p_{n+1,0}^{\mathrm{gate}}(x')] = p_{n,0}^{\mathrm{gate}}(x') \in [0,1]$, $|\mu_{n+1,f}(x')-\mu_{n,f}(x')| \le \varepsilon_f(q)$, $|p_{n+1,0}^{\mathrm{gate}}(x')-p_{n,0}^{\mathrm{gate}}(x')| \le \varepsilon_g(q)$, and $(\hat{J}_n^{\mathrm{ref}} - \mu_{n,f}(x'))_+ \le B_f$, single-slot utility $V_{n+1}(x')$ satisfies $0 \le \E_n[\max_{x' \in \mathcal{C}_n^{\mathrm{eval}}} V_{n+1}(x')] - \max_{x' \in \mathcal{C}_n^{\mathrm{eval}}} V_n(x') \le \varepsilon_f(q) + B_f\varepsilon_g(q)$.
\end{enumerate}
\end{theorem}

\paragraph{Distinction between the menu proxy and actual terminal nomination.}
Parts~1--2 of Theorem~\ref{thm:eckg_dual_voi} follow from the inclusion $\mathcal{S}_n^\star \in \mathfrak{M}_0^{(s)} \subseteq \mathfrak{M}_1^{(s)}(q)$ under menu maximization. Because actual terminal clean-up invokes the nominator once to form $\hat{\mathcal{S}}_N(\mathcal{D}_N)$ rather than maximizing posterior utility over $\{\varnothing, \mathcal{S}_{N-1}^\star, \hat{\mathcal{S}}_N(\mathcal{D}_N)\}$, the frozen working model assigns the single-nomination gain $\Delta_n^{\mathrm{term}}(q) = \E_n[U_{n+1}(\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n \cup \{q\})) - U_{n+1}(\hat{\mathcal{S}}_n(\mathcal{D}_n))]$, which need not be nonnegative and can rank high-variance shortlist perturbations differently from $\widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q)$ (for example, a query producing post-query shortlist utilities $0.5$ or $1.5$ with equal probability around baseline $1.0$ has $\Delta_n^{\mathrm{term}} = 0$ but menu proxy $0.25$). Retaining $\mathcal{S}_n^\star$ inside $\mathfrak{M}_0^{(s)}$ keeps the pre-query nomination available in every fantasy, and we evaluate query rankings directly under $\Delta_n^{\mathrm{term}}$ on frozen states in Section~\ref{sec:results} (see also Proposition~\ref*{prop:cascade_rank} in Section~\ref*{ec:cascade_rank} for single-slot vs.\ cascade ranking margins).

\begin{theorem}[Distribution-Free Familywise Safety of the Guarded Clean-Up Cascade]\label{thm:vkg_safety}
Condition on any frozen shortlist $\mathcal{S} = (x_{(0)}, \dots, x_{(L-1)})$ ($1 \le L \le 3$) and search history $\mathcal{F}_N$. Suppose verification constraint indicators at each slot $i$ are i.i.d.\ Bernoulli($p_t^{\mathrm{ver}}(x_{(i)})$) under verification law $\mathbb{Q}_t^{\mathrm{ver}}$, and each slot applies the exact lower-tail binomial gate at risk threshold $\alpha$ and level $\delta_i \ge 0$ with $\sum_{i=0}^{L-1} \delta_i \le \delta$. Then under both the incumbent-guarded branch and the sequential first-pass branch,
\begin{equation}\label{eq:fwer_safety}
  \Prob_{\mathbb{Q}_t^{\mathrm{ver}}}\!\big(\mathrm{state} = \texttt{certified\_deployment} \;\text{and}\; p_t^{\mathrm{ver}}(\hat{x}_t) > \alpha \,\big|\, \mathcal{F}_N\big) \;\le\; \delta,
\end{equation}
using at most $B_{\mathrm{ver}} = \MaxVerifCalls$ verification replications ($k^\star = 1$ across $(\SlotZeroK, \SlotOneK, \SlotTwoK)$ at $\alpha = 0.05, \delta = 0.05$).
\end{theorem}

\paragraph{Archive Call Amortization.}
When historical trajectories are pre-existing sunk enterprise logs ($B_{\mathrm{src}}^{\mathrm{marginal}} = 0$), any per-task call reduction $\Delta B = B_{\mathrm{cold}} - B_{\mathrm{FTA}} > 0$ at equal or lower deployment loss yields immediate savings ($M_{\mathrm{break}}^{\mathrm{sunk}} = 1$); if $B_{\mathrm{src}} > 0$ offline source calls are generated specifically to build the archive across $M$ target tasks, full simulation cost breaks even at $M \ge M_{\mathrm{break}}^{\mathrm{full}} = \lceil B_{\mathrm{src}} / \Delta B \rceil$ (reported separately from post-hoc research audit calls in Table~\ref*{tab:amortization}).
