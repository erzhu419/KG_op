\section{Theoretical Analysis}\label{sec:theory}

We establish four formal results governing occupancy-balanced subspace compression, shift-invariant variance and transfer dominance, self-normalized \texttt{SURE} uniform shift excess risk, and dual Eligible-Cascade VOI decomposition alongside finite-sample clean-up safety. Core algebraic and measure-theoretic statements are machine-checked in Lean~4.31.0 (Section~\ref*{ec:lean_map}).

\subsection{Self-Normalized Occupancy Balancing, Shift Invariance, and Transfer Dominance}

Let $\nu$ denote a reference probability measure on exposure domain $\mathcal{M}_\psi$ and $\pi_{\mathrm{occ}} > 0$ the empirical occupancy distribution induced by adaptive source solvers, with density ratio $\rho(\psi) = \nu(\psi)/\pi_{\mathrm{occ}}(\psi)$.

\begin{proposition}[Self-Normalized Clipping Bias and Parseval Separation]\label{prop:subspace}
Let clipped weights $\tilde{w}(\psi) = \min\{\bar{w}, \rho(\psi)\}$ define both the unnormalized measure $\nu_{\bar{w}}(\mathrm{d}\psi) = \tilde{w}(\psi)\pi_{\mathrm{occ}}(\mathrm{d}\psi)$ and the self-normalized probability measure $\tilde{\nu}_{\bar{w}}(\mathrm{d}\psi) = \nu_{\bar{w}}(\mathrm{d}\psi)/Z_{\bar{w}}$ with $Z_{\bar{w}} = \int_{\mathcal{M}_\psi} \nu_{\bar{w}}(\mathrm{d}\psi) > 0$ (matching~\eqref{eq:inv_occ_weights}). Then:
\begin{enumerate}
  \item \emph{(Unnormalized and Self-Normalized Clipping Bias)} For any $f, h \in L^2(\nu)$, $|\langle f, h \rangle_{\nu_{\bar{w}}} - \langle f, h \rangle_\nu| \le \int_{\mathcal{M}_\psi} \max\{0, \rho(\psi) - \bar{w}\}|f(\psi)h(\psi)|\pi_{\mathrm{occ}}(\mathrm{d}\psi)$. Moreover, whenever $|f(\psi)h(\psi)| \le B$ almost everywhere, the normalizing denominator $Z_{\bar{w}}$ cancels algebraically to yield the denominator-free self-normalized bound
  \begin{equation}\label{eq:norm_clip_bias}
    \big|\langle f, h \rangle_{\tilde{\nu}_{\bar{w}}} - \langle f, h \rangle_\nu\big| \;\le\; 2 B \int_{\mathcal{M}_\psi} \max\{0,\, \rho(\psi) - \bar{w}\}\,\pi_{\mathrm{occ}}(\mathrm{d}\psi),
  \end{equation}
  vanishing identically whenever $\sup_\psi \rho(\psi) \le \bar{w}$.
  \item \emph{(Active--Pruned Orthonormal Parseval Separation)} For any $\tilde{\nu}_{\bar{w}}$-orthonormal dictionary $\{\tilde{\varphi}_k\}_{k=1}^{M_+}$ partitioned into active coordinates $S_{\mathrm{act}}$ ($|S_{\mathrm{act}}| = K_{\mathrm{act}}$) and pruned coordinates $S_{\mathrm{act}}^c$, any $h(x) = \theta_0 + \sum_{k=1}^{M_+} \theta_k \tilde{\varphi}_k(\psi(x)) + r_{\mathrm{hf}}(x)$ with $r_{\mathrm{hf}} \perp \operatorname{span}(1, \tilde{\varphi})$ satisfies $\|h - \theta_0\|_{L^2(\tilde{\nu}_{\bar{w}})}^2 = \sum_{j \in S_{\mathrm{act}}} \theta_j^2 + \sum_{k \in S_{\mathrm{act}}^c} \theta_k^2 + \|r_{\mathrm{hf}}\|_{L^2(\tilde{\nu}_{\bar{w}})}^2$.
\end{enumerate}
\end{proposition}

\begin{theorem}[Within-Policy Shift Invariance and Exact Transfer Dominance Boundary]\label{thm:transfer_boundary}
Consider evaluating policy $x$ on target task $t$ where $Y_{t,g,\ell}(x) = \mu_{0,g}(x) + \Delta_g(x) + \varepsilon_\ell$ ($\ell = 1,\dots,r$) with independent zero-mean errors $\E[\varepsilon_\ell]=0$ and $\E[\varepsilon_\ell^2] = v_t(x)$.
\begin{enumerate}
  \item \emph{(Mean-Shift Contamination vs.\ Within-Policy Invariance)} Raw prior residuals satisfy $\E[(Y_{t,g,\ell}(x) - \mu_{0,g}(x))^2] = \Delta_g(x)^2 + v_t(x)$, whereas $S_{t,g}^2(x) = \sum_{\ell=1}^r (Y_{t,g,\ell}(x) - \bar{Y}_{t,g}(x))^2 = \sum_{\ell=1}^r (\varepsilon_\ell - \bar{\varepsilon})^2$ cancels $\mu_{0,g}(x) + \Delta_g(x)$ deterministically on every sample path and satisfies $\E[S_{t,g}^2(x)/(r-1)] = v_t(x)$ for any fixed batch size $r \ge 2$.
  \item \emph{(Finite-$M_{\mathrm{src}}$ Mismatch Decomposition)} For a whitened coordinate where $\hat{\mu}_0 = \mu^\star + \xi_{\mathrm{meta}} + b_{\mathrm{occ}}$ ($\E[\xi_{\mathrm{meta}}]=0$, $\Var(\xi_{\mathrm{meta}}) = \sigma_{\mathrm{src}}^2 / M_{\mathrm{src}}$) and $\theta_t = \mu^\star + \Delta_{\mathrm{shift}}$, the expected squared prior discrepancy $\Delta_{\mathrm{eff}}^2 \coloneqq (\theta_t - \hat{\mu}_0)^2$ satisfies $\E[\Delta_{\mathrm{eff}}^2] = (\Delta_{\mathrm{shift}} - b_{\mathrm{occ}})^2 + \sigma_{\mathrm{src}}^2 / M_{\mathrm{src}}$.
  \item \emph{(Exact Necessary and Sufficient Transfer Boundary)} Conditioning on $\hat{\mu}_0$ and combining prior precision $\kappa_0 > 0$ with an unbiased target estimator of precision $\rho_n > 0$ yields $\operatorname{MSE}(\kappa_0; \Delta_{\mathrm{eff}}) = (\kappa_0^2\Delta_{\mathrm{eff}}^2 + \rho_n)/(\kappa_0 + \rho_n)^2$. Tempered transfer strictly dominates target-only estimation ($\rho_n^{-1}$) \textbf{if and only if}
  \begin{equation}\label{eq:transfer_iff}
    \operatorname{MSE}(\kappa_0; \Delta_{\mathrm{eff}}) < \frac{1}{\rho_n}
    \qquad \Longleftrightarrow \qquad
    \Delta_{\mathrm{eff}}^2 < \frac{1}{\rho_n} + \frac{2}{\kappa_0}.
  \end{equation}
\end{enumerate}
\end{theorem}

While deterministic mean cancellation holds on any pooled sample, adaptive stopping or data-dependent revisiting can introduce finite-sample bias when pooling across sequential batches (e.g., two symmetric $\{-1,1\}$ draws followed by two additional draws only upon initial disagreement yield $\E[\hat{s}_{\mathrm{pool}}^2] = 7/12 < 1$; see Section~\ref*{ec:proofs}). We therefore audit both fixed-batch and pooled variance MSEs empirically on matched target paths.

\subsection{Self-Normalized Fisher-Whitened \texttt{SURE} Uniform Shift Excess-Risk Bound}

Theorem~\ref{thm:transfer_boundary}(3) shows that fixed-weight transfer ($\kappa_0 > 0$) incurs unbounded excess risk once $\Delta_{\mathrm{eff}}^2 > \rho_n^{-1} + 2/\kappa_0$. Under the Fisher-eigenspace whitening~\eqref{eq:fisher_whitening}--\eqref{eq:sure_closed_form} implemented in \textbf{GP-FTA-ECKG}, each identifiable coordinate $k \in \mathcal{I}_{n,c}$ ($|\mathcal{I}_{n,c}| = m_{\mathrm{id}} \ge 1$) satisfies $\tilde{R}_k = \delta_k + \xi_k$ with unknown shift $\delta_k$ and noise proxy $\sigma_k^2 \ge 0$ ($\sigma_k^2 \equiv 1$ in~\eqref{eq:sure_closed_form}). For shrinkage weight $\lambda \in [0, 1]$ ($\lambda=0$ pure source prior; $\lambda=1$ full target release), compare true normalized estimation loss $L_{\mathcal{I}}(\lambda) = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} ((1 - \lambda)\delta_k - \lambda \xi_k)^2$ with observable Stein--Mallows risk $\widehat{R}_{\mathcal{I}}(\lambda) = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \big[(1 - \lambda)^2 \tilde{R}_k^2 + (2\lambda - 1)\sigma_k^2\big]$.

\begin{theorem}[Self-Normalized \texttt{SURE} Uniform Shift Excess-Risk Bound]\label{thm:sure_uniform}
Let $\hat{\lambda} \in [0, 1]$ be \emph{any} empirical shrinkage/release selector (including closed-form~\eqref{eq:sure_closed_form} over $[0,1]$ or minimization over any grid $\mathcal{G} \ni 1$) satisfying $\widehat{R}_{\mathcal{I}}(\hat{\lambda}) \le \widehat{R}_{\mathcal{I}}(1)$. Write $\bar{\delta}^2 = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \delta_k^2$, $\bar{C} = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \delta_k \xi_k$, and $\bar{\sigma}^2 = \frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \sigma_k^2$. Then:
\begin{enumerate}
  \item \emph{(Self-Normalizing Quadratic Identity)} Setting $u = 1 - \lambda$, for all $\lambda \in \R$,
  \begin{equation}\label{eq:sure_identity}
    \big(L_{\mathcal{I}}(\lambda) - L_{\mathcal{I}}(1)\big) - \frac{3}{2}\big(\widehat{R}_{\mathcal{I}}(\lambda) - \widehat{R}_{\mathcal{I}}(1)\big)
    = -\frac{1}{2}u^2 \bar{\delta}^2 - u(2 + u)\bar{C} - \Big(2u + \frac{1}{2}u^2\Big)\frac{1}{m_{\mathrm{id}}}\sum_{k \in \mathcal{I}} \xi_k^2 + 3u\bar{\sigma}^2.
  \end{equation}
  \item \emph{(Uniform Sample-Path Excess-Loss Bound Over All Shifts $\bm{\delta} \in \R^{m_{\mathrm{id}}}$)} If $\bar{\delta}^2 = 0$, $L_{\mathcal{I}}(\hat{\lambda}) - L_{\mathcal{I}}(1) \le 3\bar{\sigma}^2$. For \emph{every} nonzero shift $\bm{\delta} \neq 0$—including $\|\bm{\delta}\|_2 \to \infty$ and with zero dependence on $|\mathcal{G}|$—
  \begin{equation}\label{eq:sure_uniform_path}
    L_{\mathcal{I}}(\hat{\lambda}) - L_{\mathcal{I}}(1) \;\le\; 3\bar{\sigma}^2 + \frac{9\bar{C}^2}{2\bar{\delta}^2},
  \end{equation}
  which reduces to $3 + 9\bar{C}^2/(2\bar{\delta}^2)$ under exact Fisher whitening $\sigma_k^2 \equiv 1$ (\texttt{fisher\_whitened\_unit\_variance\_sure\_bound}) and is homogeneous of degree $0$ in $\|\bm{\delta}\|_2$.
  \item \emph{(Uniform Expected Excess Risk)} Whenever $\{\xi_k\}_{k \in \mathcal{I}}$ are pairwise orthogonal with $\E[\xi_k^2] \le \sigma_{\max}^2$ and $\sigma_k^2 \le \sigma_{\max}^2$ (conditional on fixed $W_{n,c}$, $\sigma_{\max}^2 = 1$), $\sup_{\bm{\delta} \in \R^{m_{\mathrm{id}}}} \E[L_{\mathcal{I}}(\hat{\lambda}) - L_{\mathcal{I}}(1)] \le (3 + \frac{9}{2 m_{\mathrm{id}}})\sigma_{\max}^2$ in whitened units—equivalently $O(\Lambda_{n,c,\min}^{-1})$ in unwhitened parameter MSE as $\Lambda_{n,c,\min} \to \infty$.
\end{enumerate}
\end{theorem}

\subsection{Dual Eligible-Cascade VOI Decomposition and Clean-Up Safety}\label{sec:cascade_value}

\begin{theorem}[Dual Eligible-Cascade VOI Decomposition and Single-Slot Innovation Cap]\label{thm:eckg_dual_voi}\label{thm:sequential_bgpkg}
Fix search state $\mathcal{F}_n$ with evaluated policy set $\mathcal{D}_n$, non-negative fantasy weights $\{w_s\}_{s=1}^{S_{\mathrm{out}}}$ ($\sum_s w_s = 1$), post-query conditional cascade utility $U_{n+1}(\mathcal{S}; q, \xi^{(s)})$ ($U_{n+1}(\varnothing) \equiv 0$), current shortlist $\mathcal{S}_n^\star$, and nested menus $\mathfrak{M}_0^{(s)} = \{\varnothing, \mathcal{S}_n^\star, \hat{\mathcal{S}}_{n+1}(\mathcal{D}_n)\} \subseteq \mathfrak{M}_1^{(s)}(q) = \mathfrak{M}_0^{(s)} \cup \{\hat{\mathcal{S}}_{n+1}(\mathcal{D}_n \cup \{q\})\}$. Then:
\begin{enumerate}
  \item \emph{(Exact Dual VOI Decomposition and Non-Negativity)} For every candidate query $q \in \mathcal{C}_n$, $\widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q) = \widehat{\mathrm{VOI}}_n^{\mathrm{refine}}(q) + \widehat{\mathrm{VOI}}_n^{\mathrm{admit}}(q)$ with $\widehat{\mathrm{VOI}}_n^{\mathrm{refine}}(q) \ge 0$, $\widehat{\mathrm{VOI}}_n^{\mathrm{admit}}(q) \ge 0$, and $\widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q) \ge 0$.
  \item \emph{(Revisit vs.\ Admission Dichotomy)} For every already-evaluated revisit $q \in \mathcal{D}_n$, $\mathcal{D}_n \cup \{q\} = \mathcal{D}_n$ implies $\widehat{\mathrm{VOI}}_n^{\mathrm{admit}}(q) = 0$ identically, so $\widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q) = \widehat{\mathrm{VOI}}_n^{\mathrm{refine}}(q)$; if in addition $q \in \mathcal{D}_n$ induces zero posterior innovation, $\widehat{\mathrm{VOI}}_n^{\mathrm{EC}}(q) = 0$.
  \item \emph{(Single-Slot Submartingale and Lipschitz Cap)} On a factored single-slot evaluation set $\mathcal{C}_n^{\mathrm{eval}}$ where $\E_n[\mu_{n+1,f}(x')] = \mu_{n,f}(x')$, $\E_n[p_{n+1,0}^{\mathrm{gate}}(x')] = p_{n,0}^{\mathrm{gate}}(x') \in [0,1]$, $|\mu_{n+1,f}(x')-\mu_{n,f}(x')| \le \varepsilon_f(q)$, $|p_{n+1,0}^{\mathrm{gate}}(x')-p_{n,0}^{\mathrm{gate}}(x')| \le \varepsilon_g(q)$, and $(\hat{J}_n^{\mathrm{ref}} - \mu_{n,f}(x'))_+ \le B_f$, single-slot utility $V_{n+1}(x')$ is a submartingale ($\E_n[V_{n+1}(x')] \ge V_n(x')$) satisfying $0 \le \E_n[\max_{x' \in \mathcal{C}_n^{\mathrm{eval}}} V_{n+1}(x')] - \max_{x' \in \mathcal{C}_n^{\mathrm{eval}}} V_n(x') \le \varepsilon_f(q) + B_f\varepsilon_g(q)$, preventing greedy starvation of positive-value candidates as $\varepsilon_f(q), \varepsilon_g(q) \to 0$.
\end{enumerate}
\end{theorem}

Electronic Companion Section~\ref*{ec:cascade_rank} establishes a sufficient margin condition (Proposition~\ref*{prop:cascade_rank}) characterizing when single-slot and full-cascade query rankings coincide via branchwise guard and fallback discrepancy bounds.

\begin{theorem}[Distribution-Free Familywise Safety of the Guarded Clean-Up Cascade]\label{thm:vkg_safety}
Condition on any frozen shortlist $\mathcal{S} = (x_{(0)}, \dots, x_{(L-1)})$ ($1 \le L \le 3$) and search history $\mathcal{F}_N$. Suppose verification constraint indicators at each slot $i$ are i.i.d.\ Bernoulli($p_t^{\mathrm{ver}}(x_{(i)})$) under verification law $\mathbb{Q}_t^{\mathrm{ver}}$, and each slot applies the exact lower-tail binomial gate at risk threshold $\alpha$ and level $\delta_i \ge 0$ with $\sum_{i=0}^{L-1} \delta_i \le \delta$. Then under both the incumbent-guarded branch and the sequential first-pass branch,
\begin{equation}\label{eq:fwer_safety}
  \Prob_{\mathbb{Q}_t^{\mathrm{ver}}}\!\big(\mathrm{state} = \texttt{certified\_deployment} \;\text{and}\; p_t^{\mathrm{ver}}(\hat{x}_t) > \alpha \,\big|\, \mathcal{F}_N\big) \;\le\; \delta,
\end{equation}
using at most $B_{\mathrm{ver}} = \MaxVerifCalls$ verification replications ($k^\star = 1$ across $(\SlotZeroK, \SlotOneK, \SlotTwoK)$ at $\alpha = 0.05, \delta = 0.05$).
\end{theorem}

\paragraph{Remark (Archive Call Amortization).}
When historical trajectories are pre-existing sunk enterprise logs ($B_{\mathrm{src}}^{\mathrm{marginal}} = 0$), any per-task call reduction $\Delta B = B_{\mathrm{cold}} - B_{\mathrm{FTA}} > 0$ at equal or lower deployment loss yields immediate savings ($M_{\mathrm{break}}^{\mathrm{sunk}} = 1$); if $B_{\mathrm{src}} > 0$ offline source calls are generated from scratch across $M$ target tasks, total simulation cost breaks even at $M \ge M_{\mathrm{break}}^{\mathrm{full}} = \lceil B_{\mathrm{src}} / \Delta B \rceil$ (evaluated empirically against both raw-space and same-representation cold-start comparators in Table~\ref*{tab:amortization}).
